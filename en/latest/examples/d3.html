d3 Integration / D3.js 矢量可视化集成

d31

Example of using OpenLayers and d3 together. / 演示如何将 OpenLayers 与 D3.js（d3-geo）深度结合，将 TopoJSON 几何渲染到 SVG 图层中。

The example loads TopoJSON geometries and uses d3 (d3.geo.path) to render these geometries to a SVG element.

本示例演示如何将 OpenLayers 强大的地理地图视口与 D3.js（d3-geo / d3-selection） 数据可视化生态无缝融合。通过继承基础 Layer 类构建自定义图层，在每帧渲染时动态将 OpenLayers 的视口分辨率、中心与旋转角度同步给 D3 的墨卡托投影投影器（d3.geoMercator），从而使用 d3.geoPath 将 TopoJSON 拓扑几何精准绘制在 SVG 画布上。

📚 学习记录与技术深度解读 (Learning Notes)

1. 核心架构与设计思想 (Core Architecture & Design)

  • OpenLayers 与 D3.js 的架构互补性：
    • OpenLayers 专精于全尺度地图切片、坐标投影系统、视口缩放平移交互与空间数据源加载；
    • D3.js 专精于数据驱动文档（Data-Driven Documents）、微观统计图表、精美贝塞尔连线与 SVG 动画过渡。
    • 通过将 D3 的 geoPath 包装在 OpenLayers 自定义图层的 render 生命周期中，既保留了瓦片底图的高速平移缩放体验，又获得了 D3 强大的矢量拓扑数据表达能力。
  • 视图状态双向对齐算法： D3 的投影器拥有自身独立的缩放（scale）、中心点（center）、位移（translate）与旋转（angle）。OpenLayers 通过比较要素包围盒在地理空间下的实际跨度与在 D3 像素空间下的尺寸比例，动态解算出当前的绝对 scale = r / frameState.viewState.resolution，实现 100% 像素级贴合。

2. 源码与关键 API 剖析 (Source Code & API Analysis)

  • 自定义 CanvasLayer extends Layer：
    • getSourceState()：返回 'ready'，通知地图管线当前图层资源已就绪，无需等待外部数据源。
    • render(frameState)：核心帧回调。从 frameState 中提取当前视口宽高 size、地理中心 getCenter(extent) 与旋转角度 rotation，并实时调用 d3Projection.scale(...).center(...).translate(...).angle(...)。
  • topojson-client 与 d3.geoPath： 利用 feature(us, 'counties') 将高度压缩的 TopoJSON 拓扑弧段无损展开为 GeoJSON 要素集合，直接喂给 d3Path 生成 SVG d 路径字符串。

3. 生产实践踩坑与性能调优 (Production Pitfalls & Optimization)

  • 连续缩放时的 DOM 更新开销： 每当视口移动或缩放一帧，D3 都需要对成千上万个县级边界多边形重新执行投影数学计算并生成巨大的 SVG d 属性字符串，可能导致大范围动画时丢帧。生产中若多边形顶点极多，推荐将 D3 改为渲染到 <canvas> 上下文（geoPath().context(ctx)）以提升渲染帧率。
  • 坐标系转换单位规范： D3 的 d3.geoMercator() 默认接受经纬度 [lon, lat]。因此必须使用 toLonLat(getCenter(extent), projection) 将 OpenLayers 内部的 3857 投影坐标逆转为经纬度后再传给 D3 的 center()。
main.js 
import {json} from 'd3-fetch';
import {geoBounds, geoMercator, geoPath} from 'd3-geo';
import {select} from 'd3-selection';
import Map from 'ol/Map.js';
import View from 'ol/View.js';
import {getCenter, getWidth} from 'ol/extent.js';
import Layer from 'ol/layer/Layer.js';
import TileLayer from 'ol/layer/Tile.js';
import {fromLonLat, toLonLat} from 'ol/proj.js';
import StadiaMaps from 'ol/source/StadiaMaps.js';
import {feature} from 'topojson-client';

/**
 * Custom OpenLayers Layer integrating D3.js SVG rendering pipeline
 * 自定义 OpenLayers 图层类，集成 D3.js 的 SVG 渲染管线
 */
class CanvasLayer extends Layer {
  constructor(options) {
    super(options);

    this.features = options.features;

    // Create an absolute positioned SVG DOM node wrapped with D3 selection
    // 创建一个通过 D3 选择器包装的绝对定位 SVG DOM 节点
    this.svg = select(document.createElement('div'))
      .append('svg')
      .style('position', 'absolute');

    this.svg.append('path').datum(this.features).attr('class', 'boundary');
  }

  getSourceState() {
    return 'ready';
  }

  /**
   * Render function called on each map render frame
   * 每帧地图渲染时触发的核心渲染函数，同步 OpenLayers 视口状态到 D3 投影器
   * @param {import('ol/Map.js').FrameState} frameState Frame state / 帧渲染状态
   * @return {HTMLElement} SVG DOM node / SVG 根节点
   */
  render(frameState) {
    const width = frameState.size[0];
    const height = frameState.size[1];
    const projection = frameState.viewState.projection;
    const d3Projection = geoMercator().scale(1).translate([0, 0]);
    let d3Path = geoPath().projection(d3Projection);

    // Calculate pixel bounds in D3 base projection
    // 在 D3 基准投影空间中计算要素的像素包围盒
    const pixelBounds = d3Path.bounds(this.features);
    const pixelBoundsWidth = pixelBounds[1][0] - pixelBounds[0][0];
    const pixelBoundsHeight = pixelBounds[1][1] - pixelBounds[0][1];

    // Calculate geographical bounds in OpenLayers projection
    // 在 OpenLayers 地图投影坐标系中计算要素的实际空间包围盒
    const bounds = geoBounds(this.features);
    const geoBoundsLeftBottom = fromLonLat(bounds[0], projection);
    const geoBoundsRightTop = fromLonLat(bounds[1], projection);
    let geoBoundsWidth = geoBoundsRightTop[0] - geoBoundsLeftBottom[0];
    if (geoBoundsWidth < 0) {
      geoBoundsWidth += getWidth(projection.getExtent());
    }
    const geoBoundsHeight = geoBoundsRightTop[1] - geoBoundsLeftBottom[1];

    // Compute scale factor matching current view resolution
    // 计算匹配当前视口分辨率的缩放因子
    const widthResolution = geoBoundsWidth / pixelBoundsWidth;
    const heightResolution = geoBoundsHeight / pixelBoundsHeight;
    const r = Math.max(widthResolution, heightResolution);
    const scale = r / frameState.viewState.resolution;

    // Convert view center to longitude/latitude and rotation to degrees
    // 将视口中心转为经纬度，并将旋转弧度换算为角度
    const center = toLonLat(getCenter(frameState.extent), projection);
    const angle = (-frameState.viewState.rotation * 180) / Math.PI;

    // Synchronize D3 Mercator projection parameters with OpenLayers view state
    // 将 OpenLayers 视口参数注入 D3 墨卡托投影器中
    d3Projection
      .scale(scale)
      .center(center)
      .translate([width / 2, height / 2])
      .angle(angle);

    d3Path = d3Path.projection(d3Projection);
    d3Path(this.features);

    this.svg.attr('width', width);
    this.svg.attr('height', height);

    this.svg.select('path').attr('d', d3Path);

    return this.svg.node();
  }
}

// Main map instance with Stamen watercolor basemap
// 地图主实例，加载 Stamen 水彩底图
const map = new Map({
  layers: [
    new TileLayer({
      source: new StadiaMaps({
        layer: 'stamen_watercolor',
      }),
    }),
  ],
  target: 'map',
  view: new View({
    center: fromLonLat([-97, 38]),
    zoom: 4,
  }),
});

/**
 * Load TopoJSON data and instantiate the custom D3 layer
 * 加载 TopoJSON 数据并实例化自定义 D3 SVG 图层挂载至地图
 */
json('https://cdn.jsdelivr.net/gh/gisnotes/openlayers@gh-pages/examples/data/topojson/us.json').then(function (us) {
  const layer = new CanvasLayer({
    features: feature(us, 'counties'),
  });

  map.addLayer(layer);
});
