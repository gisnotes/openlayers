Demonstrates color manipulation with a raster source. / 演示利用 RasterSource 在 Web Worker 中进行像素级色相、色度、亮度（HCL）动态色彩空间变换。

A raster source allows arbitrary manipulation of pixel values. In this example, RGB values on the input tile source are adjusted in a pixel-wise operation before being rendered with a second raster source. The raster operation takes pixels in RGB space, converts them to HCL color space, adjusts the values based on the controls above, and then converts them back to RGB space for rendering.

栅格数据源（RasterSource）允许开发者在客户端对任意图层的像元像素进行深度数学运算。在这一示例中，输入的底图切片源像素在被渲染之前，由后台独立 Web Worker 执行逐像素数学变换：将原生 RGB 颜色空间映射到更加贴合人类视觉感知的 HCL（Hue 色相, Chroma 色度, Lightness 亮度） 色彩空间，根据界面控件动态调整对应分量后，再逆变换回 RGB 空间渲染输出。

--- ### 📚 学习记录与技术深度解读 (Learning Notes) #### 1. 核心架构与设计思想 (Core Architecture & Design) - `RasterSource` 客户端栅格代数计算管线： 传统的地图滤镜大多基于 CSS `filter`（如 `brightness()`、`contrast()`），其调整粗糙且无法针对非线性色彩空间操作。`RasterSource` 在独立于 UI 主线程的 Web Worker 中接收输入图层（如 `StadiaMaps` 水彩底图）的原始 `ImageData` 像素流，通过并行像元循环执行像素重构，彻底杜绝主线程卡顿。 - 为什么要从 RGB 转换到 HCL 色彩空间？： RGB 空间是面向硬件发光管（红绿蓝）设计的，其三个通道在数学上与人类的主观明暗感知高度非线性耦合（例如绿色发光强度远高于蓝色）。而 **HCL（基于 CIE L\*a\*b\* 柱面极坐标） 具有严格的感知均匀性（Perceptually Uniform）**： - 调节 `Lightness` 只改变明暗，完全不影响色相； - 调节 `Hue` 在色环上均匀旋转，不会像 HSV 一样在黄色区域产生刺眼的假亮； - 调节 `Chroma` 线性改变色彩饱和纯度。

2. 源码与关键 API 剖析 (Source Code & API Analysis) - RasterSource({ sources, operation, lib })：

  • sources：输入图层源数组（支持多个源同时输入进行波段代数运算）。
  • operation(pixels, data)：每个像元执行的核心算法回调函数。必须保持为纯函数（无外部闭包变量引用）。
  • lib：将 Worker 所需的纯数学辅助函数（如 rgb2hcl、hcl2rgb）序列化后注入 Worker 运行上下文。
  • beforeoperations 数据注入与 raster.changed() 刷新通知： 监听 raster.on('beforeoperations', (evt) => { evt.data.hue = ... })，将 UI 主线程当前的滑动条数值打包写入 Worker 每帧执行上下文；滑块滑动时调用 raster.changed() 立即触发新一轮 Worker 运算。

3. 生产实践踩坑与性能调优 (Production Pitfalls & Optimization) - Worker 函数作用域隔离（lib 陷阱）：

operation 函数内部绝不能访问主线程的 window、document 或模块闭包作用域变量。所有依赖的数学常量（如 twoPi、Xn）和辅助函数必须显式列在 lib 属性字典中，否则在 Worker 线程解析时会报 ReferenceError 致命异常。

  • 跨域切片 CORS 限制： RasterSource 依赖 ctx.getImageData() 提取图像像素。如果底图瓦片服务器未正确配置 CORS 跨域响应头（Access-Control-Allow-Origin: *），浏览器会直接判定 Canvas 被污染（Tainted Canvas）并抛出安全安全错误拒绝导出像素。
main.js 
import Map from 'ol/Map.js';
import View from 'ol/View.js';
import ImageLayer from 'ol/layer/Image.js';
import RasterSource from 'ol/source/Raster.js';
import StadiaMaps from 'ol/source/StadiaMaps.js';

/**
 * Color manipulation functions below are adapted from https://github.com/d3/d3-color
 * 以下颜色空间转换数学公式改编自 d3-color 开源算法库
 */
const Xn = 0.95047;
const Yn = 1;
const Zn = 1.08883;
const t0 = 4 / 29;
const t1 = 6 / 29;
const t2 = 3 * t1 * t1;
const t3 = t1 * t1 * t1;
const twoPi = 2 * Math.PI;

/**
 * Convert an RGB pixel into an HCL pixel
 * 将 RGB 像素转换为感知均匀的 HCL 色彩空间像素
 * @param {Array<number>} pixel A pixel in RGB space / RGB 像素通道值数组
 * @return {Array<number>} A pixel in HCL space / HCL 像素通道值数组
 */
function rgb2hcl(pixel) {
  const red = rgb2xyz(pixel[0]);
  const green = rgb2xyz(pixel[1]);
  const blue = rgb2xyz(pixel[2]);

  const x = xyz2lab(
    (0.4124564 * red + 0.3575761 * green + 0.1804375 * blue) / Xn,
  );
  const y = xyz2lab(
    (0.2126729 * red + 0.7151522 * green + 0.072175 * blue) / Yn,
  );
  const z = xyz2lab(
    (0.0193339 * red + 0.119192 * green + 0.9503041 * blue) / Zn,
  );

  const l = 116 * y - 16;
  const a = 500 * (x - y);
  const b = 200 * (y - z);

  const c = Math.sqrt(a * a + b * b);
  let h = Math.atan2(b, a);
  if (h < 0) {
    h += twoPi;
  }

  pixel[0] = h;
  pixel[1] = c;
  pixel[2] = l;

  return pixel;
}

/**
 * Convert an HCL pixel into an RGB pixel
 * 将 HCL 空间像素反向映射回用于硬件显示的 RGB 像素
 * @param {Array<number>} pixel A pixel in HCL space / HCL 像素通道值数组
 * @return {Array<number>} A pixel in RGB space / RGB 像素通道值数组
 */
function hcl2rgb(pixel) {
  const h = pixel[0];
  const c = pixel[1];
  const l = pixel[2];

  const a = Math.cos(h) * c;
  const b = Math.sin(h) * c;

  let y = (l + 16) / 116;
  let x = isNaN(a) ? y : y + a / 500;
  let z = isNaN(b) ? y : y - b / 200;

  y = Yn * lab2xyz(y);
  x = Xn * lab2xyz(x);
  z = Zn * lab2xyz(z);

  pixel[0] = xyz2rgb(3.2404542 * x - 1.5371385 * y - 0.4985314 * z);
  pixel[1] = xyz2rgb(-0.969266 * x + 1.8760108 * y + 0.041556 * z);
  pixel[2] = xyz2rgb(0.0556434 * x - 0.2040259 * y + 1.0572252 * z);

  return pixel;
}

function xyz2lab(t) {
  return t > t3 ? Math.pow(t, 1 / 3) : t / t2 + t0;
}

function lab2xyz(t) {
  return t > t1 ? t * t * t : t2 * (t - t0);
}

function rgb2xyz(x) {
  return (x /= 255) <= 0.04045 ? x / 12.92 : Math.pow((x + 0.055) / 1.055, 2.4);
}

function xyz2rgb(x) {
  return (
    255 * (x <= 0.0031308 ? 12.92 * x : 1.055 * Math.pow(x, 1 / 2.4) - 0.055)
  );
}

// Create RasterSource running pixel operations in Web Worker
// 创建在后台 Web Worker 中运行像素级运算的栅格数据源
const raster = new RasterSource({
  sources: [
    new StadiaMaps({
      layer: 'stamen_watercolor',
    }),
  ],
  operation: function (pixels, data) {
    const hcl = rgb2hcl(pixels[0]);

    // Apply hue rotation
    // 应用色相旋转
    let h = hcl[0] + (Math.PI * data.hue) / 180;
    if (h < 0) {
      h += twoPi;
    } else if (h > twoPi) {
      h -= twoPi;
    }
    hcl[0] = h;

    // Apply chroma and lightness scaling
    // 应用色度与亮度比例缩放
    hcl[1] *= data.chroma / 100;
    hcl[2] *= data.lightness / 100;

    return hcl2rgb(hcl);
  },
  lib: {
    rgb2hcl: rgb2hcl,
    hcl2rgb: hcl2rgb,
    rgb2xyz: rgb2xyz,
    lab2xyz: lab2xyz,
    xyz2lab: xyz2lab,
    xyz2rgb: xyz2rgb,
    Xn: Xn,
    Yn: Yn,
    Zn: Zn,
    t0: t0,
    t1: t1,
    t2: t2,
    t3: t3,
    twoPi: twoPi,
  },
});

const controls = {};

// Send current UI slider values into worker before each operation pass
// 在每次像素运算轮次前将 UI 滑动条当前值打包传递给 Worker 线程
raster.on('beforeoperations', function (event) {
  const data = event.data;
  for (const id in controls) {
    data[id] = Number(controls[id].value);
  }
});

// Main map instance rendering the dynamic raster source
// 地图主实例，呈现经过动态像素色彩变换的栅格图层
const map = new Map({
  layers: [
    new ImageLayer({
      source: raster,
    }),
  ],
  target: 'map',
  view: new View({
    center: [0, 2500000],
    zoom: 2,
    maxZoom: 18,
  }),
});

// Wire slider event listeners to update labels and trigger raster recomputation
// 绑定滑动条监听事件以同步更新数值标注并触发栅格重算
const controlIds = ['hue', 'chroma', 'lightness'];
controlIds.forEach(function (id) {
  const control = document.getElementById(id);
  const output = document.getElementById(id + 'Out');
  control.addEventListener('input', function () {
    output.innerText = control.value;
    raster.changed();
  });
  output.innerText = control.value;
  controls[id] = control;
});
