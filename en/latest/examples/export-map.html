Map Export / 地图导出 (GeoTIFF / PNG / Worldfile)

export4 png2 geotiff5 worldfile1

Download / 下载地图

Export a map as a GeoTIFF or PNG image. / 将当前地图视口导出为带有地理参考的 GeoTIFF 文件、普通 PNG 图片或带有 Worldfile 坐标定位文件的 ZIP 压缩包。

Example of exporting a map, using a HTMLCanvasElement as temporary map target. PNG is easy to export, without using an external library. The GeoTIFF export is georeferenced with the correct projection. The PNG + worldfile ZIP is also georeferenced, but users of the image will have to know (or guess) the projection.

本示例演示如何通过将临时 HTMLCanvasElement 设定为地图渲染目标来导出地图。 导出普通 PNG 非常直接，无需依赖任何第三方外部库； 导出 GeoTIFF 则会内嵌完整的投影坐标系和地理参考参数（Tiepoint 与 PixelScale）； 导出 PNG + Worldfile（.pgw 坐标定标文件）打包为 ZIP，同样具备地理参考能力，但使用者需要自行知晓对应的投影坐标系。

📚 学习记录与技术深度解读 (Learning Notes)

1. 离屏 Canvas 重定向渲染架构

  • 传统 DOM 截图的痛点：直接截取屏幕地图可能会受到 CSS 变换、视网膜屏幕 DPR 缩放、HTML 控件（按钮、缩放滑块）遮挡等干扰。
  • map.setTarget(mapCanvas) 临时重定向模式：
    1. 动态创建一个与地图当前像素尺寸相同的离屏 <canvas> 元素；
    2. 执行 map.setTarget(mapCanvas) 将渲染管线切换至离屏画布；
    3. 监听 map.once('rendercomplete', ...) 事件，确保所有矢量图层与热力图层完全重绘完成；
    4. 提取像素字节并生成下载，最后调用 map.setTarget('map') 恢复屏幕正常展示。整个过程在几十毫秒内完成，用户视觉几乎无感知。

2. 前端纯 JavaScript 生成 GeoTIFF：geotiff.js 规范

  • 通过 geotiff 库的 writeArrayBuffer 函数，直接将 Canvas 的 RGBA 像素矩阵编码为符合 TIFF/GeoTIFF 标准的二进制字节数组：
    const tiff = writeGeotiff(imageData.data, {
  width: size[0],
  height: size[1],
  ModelPixelScale: [resolution, resolution, 0], // X 与 Y 方向的像元地面分辨率
  ModelTiepoint: [0, 0, 0, extent[0], extent[3], 0], // 左上角像元 (0,0) 对应的真实地图投影坐标 (Xmin, Ymax)
  GTRasterTypeGeoKey: 1, // 像元定位类型：RasterPixelIsArea (像元表示面积区域)
  ProjectedCSTypeGeoKey: parseInt(epsgCode), // 投影代码（如 3857）
});
  • 导出的 .tiff 文件可以直接拖入 QGIS、ArcGIS 或任何专业遥感软件，位置分毫不差！

3. Worldfile 定标文件原理 (.pgw / .tfw)

Worldfile 是一种历史悠久但兼容性极广的栅格定位文本文件（例如 map.png 对应 map.pgw）。它由 6 行纯文本数字构成：

  1. 行 1：$X$ 方向的像素分辨率；
  2. 行 2：$Y$ 轴旋转项（通常为 0）；
  3. 行 3：$X$ 轴旋转项（通常为 0）；
  4. 行 4：$Y$ 方向的像素分辨率（通常为负数，因为图像 Y 轴向下，而地理坐标 Y 轴向上）；
  5. 行 5：左上角中心像元的 $X$ 地理坐标；
  6. 行 6：左上角中心像元的 $Y$ 地理坐标。
main.js 
import {zip} from 'fflate';
import {writeArrayBuffer as writeGeotiff} from 'geotiff';
import Map from 'ol/Map.js';
import View from 'ol/View.js';
import {asArray} from 'ol/color.js';
import GeoJSON from 'ol/format/GeoJSON.js';
import HeatmapLayer from 'ol/layer/Heatmap.js';
import VectorLayer from 'ol/layer/Vector.js';
import VectorSource from 'ol/source/Vector.js';
import Fill from 'ol/style/Fill.js';
import Style from 'ol/style/Style.js';

const style = new Style({
  fill: new Fill({
    color: '#eeeeee',
  }),
});

const map = new Map({
  layers: [
    new VectorLayer({
      source: new VectorSource({
        url: 'https://openlayers.org/data/vector/ecoregions.json',
        format: new GeoJSON(),
      }),
      background: 'white',
      style: function (feature) {
        const color = asArray(feature.get('COLOR_NNH') || '#eeeeee');
        color[3] = 0.75;
        style.getFill().setColor(color);
        return style;
      },
    }),
    new HeatmapLayer({
      source: new VectorSource({
        url: 'data/geojson/world-cities.geojson',
        format: new GeoJSON(),
      }),
      weight: function (feature) {
        return feature.get('population') / 1e7;
      },
      radius: 15,
      blur: 15,
      opacity: 0.75,
    }),
  ],
  target: 'map',
  view: new View({
    center: [0, 0],
    zoom: 2,
  }),
});

// Handle export button click by redirecting map rendering to an offscreen canvas
// 点击导出按钮时，将地图渲染重定向到离屏 Canvas 画布中
document.getElementById('export-map').addEventListener('click', () => {
  const format = document.getElementById('export-format').value;
  const mapCanvas = document.createElement('canvas');
  const size = map.getSize();
  mapCanvas.width = size[0];
  mapCanvas.height = size[1];

  map.setTarget(mapCanvas);
  map.once('rendercomplete', () => {
    const view = map.getView();
    const extent = view.calculateExtent(size);
    const resolution = view.getResolution();
    const projection = view.getProjection();

    if (format === 'geotiff') {
      exportGeoTIFF(mapCanvas, size, extent, resolution, projection);
    } else if (format === 'png') {
      exportPNG(mapCanvas);
    } else if (format === 'png-world') {
      exportPNGWithWorldfile(mapCanvas, extent, resolution);
    }

    // Restore map rendering back to DOM element
    // 将地图重新挂载回 DOM 元素
    map.setTarget('map');
  });
});

// Export canvas as georeferenced GeoTIFF with projection and tiepoint metadata
// 将 Canvas 画布导出为内嵌投影和控制点元数据的 GeoTIFF 文件
function exportGeoTIFF(canvas, size, extent, resolution, projection) {
  const context = canvas.getContext('2d');
  const imageData = context.getImageData(0, 0, size[0], size[1]);
  const epsgCode = projection.getCode().split(':')[1];

  const tiff = writeGeotiff(imageData.data, {
    width: size[0],
    height: size[1],
    ModelPixelScale: [resolution, resolution, 0],
    ModelTiepoint: [0, 0, 0, extent[0], extent[3], 0],
    GTRasterTypeGeoKey: 1,
    ProjectedCSTypeGeoKey: parseInt(epsgCode),
  });

  const blob = new Blob([tiff], {type: 'image/tiff'});
  downloadFile(blob, 'map-export.tiff');
}

// Export canvas as standard PNG image
// 将 Canvas 画布导出为标准 PNG 图片
function exportPNG(canvas) {
  const link = document.createElement('a');
  link.href = canvas.toDataURL('image/png');
  link.download = 'map-export.png';
  document.body.appendChild(link);
  link.click();
  document.body.removeChild(link);
}

// Export canvas as PNG along with a 6-parameter Worldfile (.pgw) compressed into ZIP
// 将 Canvas 导出为 PNG 并附加包含 6 参数的 Worldfile (.pgw)，打包为 ZIP
function exportPNGWithWorldfile(canvas, extent, resolution) {
  // Create worldfile content (A, D, B, E, C, F affine parameters)
  // 创建 Worldfile 纯文本定标内容（A, D, B, E, C, F 仿射六参数）
  const worldfileContent = [
    resolution.toFixed(6), // pixel width / X 方向像素宽度
    '0.000000', // rotation / 旋转项
    '0.000000', // rotation / 旋转项
    (-resolution).toFixed(6), // pixel height (negative) / Y 方向像素高度（负值）
    extent[0].toFixed(6), // upper-left X / 左上角 X 坐标
    extent[3].toFixed(6), // upper-left Y / 左上角 Y 坐标
  ].join('\n');

  // Convert canvas to blob and create zip
  // 将 Canvas 转换为 Blob 并构建 ZIP 压缩文件
  canvas.toBlob((pngBlob) => {
    pngBlob.arrayBuffer().then((pngBuffer) => {
      const files = {
        'map-export.png': [new Uint8Array(pngBuffer), {level: 0}], // level 0 = no compression for PNG
        'map-export.pgw': [
          new TextEncoder().encode(worldfileContent),
          {level: 6},
        ],
      };

      zip(files, (err, data) => {
        if (err) {
          alert('Error creating zip:', err);
          return;
        }
        const zipBlob = new Blob([data], {type: 'application/zip'});
        downloadFile(zipBlob, 'map-export.zip');
      });
    });
  });
}

// Helper to trigger browser file download from Blob
// 辅助函数：触发浏览器从 Blob 下载文件
function downloadFile(blob, filename) {
  const link = document.createElement('a');
  link.href = URL.createObjectURL(blob);
  link.download = filename;
  document.body.appendChild(link);
  link.click();
  document.body.removeChild(link);
}
