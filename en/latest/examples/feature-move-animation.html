Marker Animation / 折线轨迹沿线运动与车辆巡航动效

animation5 feature12 postrender1 polyline1 maptiler26

Demonstrates how to move a feature along a line. / 演示如何使用 postrender 与 VectorContext 实现标记要素沿 Polyline 路径平滑移动与倍速调节。

This example shows how to use postrender events and a vector context to animate a marker feature along a line. In this example an encoded polyline is being used.

本示例展示如何利用 postrender 渲染完成事件与 VectorContext 矢量管线，驱动点标记沿着高压缩编码折线（Polyline）路径进行平滑连续的动态巡航。支持通过滑块实时调节运动速度倍率，且在动画进行期间将要素几何临时剥离出数据源，利用即时帧上下文绘制，实现极高帧率下的低 CPU 开销渲染。

📚 学习记录与技术深度解读 (Learning Notes)

1. 核心架构原理 (Core Architecture)

在智慧交通、车辆监控与物流路径回放系统中，车辆需要沿着数千个转折点的复杂线路行进：

  • 路径插值与高精度距离比例 (route.getCoordinateAt(fraction))：OpenLayers 的 LineString.prototype.getCoordinateAt(fraction) 提供了沿折线全长按百分比（$0.0 \sim 1.0$）线性插值空间坐标的能力，内部自动进行各分段欧几里得距离累加，避免了手动逐段解算复杂向量插值的繁琐。
  • 时间增量积分与往返巡航模运算： $$\text{distance} = \left(\text{distance} + \frac{\text{speed} \times \Delta t}{10^6}\right) \bmod 2$$ 通过对 2 取模，巧妙实现 $0 \to 1$（正向出发）与 $1 \to 2$（逆向折返，对应实际比例 $2 - \text{distance}$）的无缝连续往复巡航。
  • “临时隐藏 + 帧代理”黄金性能范式：启动动画时执行 geoMarker.setGeometry(null)，将要素实体从数据源空间索引中临时卸载；在每帧 postrender 中仅通过 vectorContext.drawGeometry(position) 将临时点绘制在视口上。停止时再通过 geoMarker.setGeometry(position) 还原。这彻底消除了每帧频繁更新 Feature 几何引发的数据源 R-Tree 空间索引重构开销！

2. 关键实现与代码拆解 (Code Deep Dive)

function moveFeature(event) {
  const speed = Number(speedInput.value);
  const time = event.frameState.time;
  const elapsedTime = time - lastTime;
  // 依据真实帧耗时增量累加位移，杜绝掉帧跳跃
  distance = (distance + (speed * elapsedTime) / 1e6) % 2;
  lastTime = time;

  // 获取当前折返位置的空间投影坐标
  const currentCoordinate = route.getCoordinateAt(
    distance > 1 ? 2 - distance : distance,
  );
  position.setCoordinates(currentCoordinate);

  // 利用帧上下文即时绘制
  const vectorContext = getVectorContext(event);
  vectorContext.setStyle(styles.geoMarker);
  vectorContext.drawGeometry(position);

  // 持续驱动下一帧重绘
  map.render();
}

3. 生产实战避坑与考量 (Production Pitfalls)

  • 车辆转向角（Heading Rotation）同步：实际车联网应用中，不仅需要更新坐标，还需要根据相邻两点差值向量计算航向角：$\theta = \operatorname{atan2}(\Delta x, \Delta y)$，并将车辆图标（Icon）的 rotation 设为 $\theta$，使车辆在拐弯时自然顺滑转向。
  • 基于帧时间增量（$\Delta t$）计算：严禁在动画中基于固定步长叠加坐标，必须使用 event.frameState.time - lastTime。当用户切换标签页或设备卡顿导致帧率波动时，基于真实时间差的积分能保证位移物理速度始终恒定。
main.js 
import Feature from 'ol/Feature.js';
import Map from 'ol/Map.js';
import View from 'ol/View.js';
import Polyline from 'ol/format/Polyline.js';
import Point from 'ol/geom/Point.js';
import TileLayer from 'ol/layer/Tile.js';
import VectorLayer from 'ol/layer/Vector.js';
import {getVectorContext} from 'ol/render.js';
import ImageTile from 'ol/source/ImageTile.js';
import VectorSource from 'ol/source/Vector.js';
import CircleStyle from 'ol/style/Circle.js';
import Fill from 'ol/style/Fill.js';
import Icon from 'ol/style/Icon.js';
import Stroke from 'ol/style/Stroke.js';
import Style from 'ol/style/Style.js';

const key = 'Get your own API key at https://www.maptiler.com/cloud/';
const attributions =
  '<a href="https://www.maptiler.com/copyright/" target="_blank">&copy; MapTiler</a> ' +
  '<a href="https://www.openstreetmap.org/copyright" target="_blank">&copy; OpenStreetMap contributors</a>';

const center = [-5639523.95, -3501274.52];
const map = new Map({
  target: document.getElementById('map'),
  view: new View({
    center: center,
    zoom: 10,
    minZoom: 2,
    maxZoom: 19,
  }),
  layers: [
    new TileLayer({
      source: new ImageTile({
        attributions: attributions,
        url: 'https://api.maptiler.com/maps/hybrid/{z}/{x}/{y}.jpg?key=' + key,
        tileSize: 512,
      }),
    }),
  ],
});

// The polyline string is read from a JSON similar to those returned
// by directions APIs such as Openrouteservice and Mapbox.
// 从类似 Openrouteservice 或 Mapbox 路径规划 API 返回的 JSON 中读取折线编码字符串
fetch('https://cdn.jsdelivr.net/gh/gisnotes/openlayers@gh-pages/examples/data/polyline/route.json').then(function (response) {
  response.json().then(function (result) {
    const polyline = result.routes[0].geometry;

    // Decode polyline to geometry (factor 1e6 for 6-decimal precision)
    // 解码 Google/Mapbox 算法压缩的折线几何
    const route = new Polyline({
      factor: 1e6,
    }).readGeometry(polyline, {
      dataProjection: 'EPSG:4326',
      featureProjection: 'EPSG:3857',
    });

    const routeFeature = new Feature({
      type: 'route',
      geometry: route,
    });
    const startMarker = new Feature({
      type: 'icon',
      geometry: new Point(route.getFirstCoordinate()),
    });
    const endMarker = new Feature({
      type: 'icon',
      geometry: new Point(route.getLastCoordinate()),
    });
    const position = startMarker.getGeometry().clone();
    const geoMarker = new Feature({
      type: 'geoMarker',
      geometry: position,
    });

    const styles = {
      'route': new Style({
        stroke: new Stroke({
          width: 6,
          color: [237, 212, 0, 0.8],
        }),
      }),
      'icon': new Style({
        image: new Icon({
          anchor: [0.5, 1],
          src: 'https://cdn.jsdelivr.net/gh/gisnotes/openlayers@gh-pages/examples/data/icon.png',
        }),
      }),
      'geoMarker': new Style({
        image: new CircleStyle({
          radius: 7,
          fill: new Fill({color: 'black'}),
          stroke: new Stroke({
            color: 'white',
            width: 2,
          }),
        }),
      }),
    };

    const vectorLayer = new VectorLayer({
      source: new VectorSource({
        features: [routeFeature, geoMarker, startMarker, endMarker],
      }),
      style: function (feature) {
        return styles[feature.get('type')];
      },
    });

    map.addLayer(vectorLayer);

    const speedInput = document.getElementById('speed');
    const startButton = document.getElementById('start-animation');
    let animating = false;
    let distance = 0;
    let lastTime;

    /**
     * Postrender animation loop: interpolate position and draw with VectorContext.
     * 渲染后帧回调：基于时间差积分计算折线插值坐标，使用 VectorContext 瞬态绘制。
     * @param {import('ol/render/Event.js').default} event Render event.
     */
    function moveFeature(event) {
      const speed = Number(speedInput.value);
      const time = event.frameState.time;
      const elapsedTime = time - lastTime;
      // Modulo 2 allows round-trip movement (0->1 forward, 1->2 backward)
      // 对 2 取模实现沿线往返往复运动（0~1 为正向，1~2 为反向折返）
      distance = (distance + (speed * elapsedTime) / 1e6) % 2;
      lastTime = time;

      const currentCoordinate = route.getCoordinateAt(
        distance > 1 ? 2 - distance : distance,
      );
      position.setCoordinates(currentCoordinate);
      const vectorContext = getVectorContext(event);
      vectorContext.setStyle(styles.geoMarker);
      vectorContext.drawGeometry(position);
      // tell OpenLayers to continue the postrender animation
      // 请求下一帧重绘继续推进动画
      map.render();
    }

    /**
     * Start animation: detach geometry from source and subscribe to postrender.
     * 启动动画：临时将 geoMarker 几何置空以优化性能，挂载 postrender 监听。
     */
    function startAnimation() {
      animating = true;
      lastTime = Date.now();
      startButton.textContent = 'Stop Animation / 停止动画';
      vectorLayer.on('postrender', moveFeature);
      // hide geoMarker and trigger map render through change event
      // 临时隐藏数据源中的要素，避免每帧触发 R-Tree 空间索引重构
      geoMarker.setGeometry(null);
    }

    /**
     * Stop animation: restore geometry to source and unsubscribe from postrender.
     * 停止动画：将最终位置赋回 geoMarker 并注销 postrender 监听。
     */
    function stopAnimation() {
      animating = false;
      startButton.textContent = 'Start Animation / 开始动画';

      // Keep marker at current animation position
      // 在当前动画停止位置还原要素几何
      geoMarker.setGeometry(position);
      vectorLayer.un('postrender', moveFeature);
    }

    startButton.addEventListener('click', function () {
      if (animating) {
        stopAnimation();
      } else {
        startAnimation();
      }
    });
  });
});
