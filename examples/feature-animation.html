Custom Animation / 要素雷达光晕扩散动画（postrender 与 VectorContext）

animation5 vector87 feature12 flash1

Demonstrates how to animate features. / 演示如何使用 postrender 渲染钩子与 VectorContext 为要素添加动态发光脉冲与波纹扩散动画。

This example shows how to use postrender and vectorContext to animate features. Here we choose to do a flash animation each time a feature is added to the layer.

本示例展示如何利用瓦片图层的 postrender 事件帧回调与 VectorContext 实时渲染管线，为空间要素附加丝滑流畅的“雷达脉冲扩散光晕（Flash Wave）”动效。每当新点要素被添加到矢量数据源时，系统会自动启动为期 3 秒的动态波纹动画：圆环半径由 5px 扩散至 30px，同时不透明度与描边线宽随时间缓动渐隐，动画结束自动注销帧监听。

📚 学习记录与技术深度解读 (Learning Notes)

1. 核心架构原理 (Core Architecture)

在传统 WebGIS 中，对每个动效要素持续修改样式或几何点坐标会频繁触发矢量源空间索引树（RBush）的重构与全局图层重绘，导致严重的性能卡顿。

  • postrender 帧钩子与直接渲染：OpenLayers 允许在图层或地图渲染流程末尾（postrender）介入。在此时刻，底图切片已经绘制完成，调用 getVectorContext(event) 可获得当前渲染帧的即时矢量画布代理（VectorContext）。
  • 无侵入的纯瞬态绘制：通过 vectorContext.drawGeometry(geom) 直接向当前帧 Canvas 上下文绘制临时动态圆，完全不污染底层 Feature 本身的几何数据与持久样式，也不触发数据源变更事件。
  • map.render() 帧循环自驱动：每次执行 animate 逻辑时，只要动画时间尚未超时（elapsed < duration），显式调用 map.render() 即可请求下一帧重绘（内部通过 requestAnimationFrame 驱动 60fps 刷新）。当达到设定周期时，通过 unByKey(listenerKey) 彻底解绑监听器，地图自然停止无谓重绘，实现零内存泄漏与最优能效。

2. 关键实现与代码拆解 (Code Deep Dive)

function flash(feature) {
  const start = Date.now();
  const flashGeom = feature.getGeometry().clone();
  // 监听底图渲染完成事件
  const listenerKey = tileLayer.on('postrender', animate);

  function animate(event) {
    const frameState = event.frameState;
    const elapsed = frameState.time - start;
    if (elapsed >= duration) {
      unByKey(listenerKey); // 动画超时，解绑注销
      return;
    }
    const vectorContext = getVectorContext(event);
    const elapsedRatio = elapsed / duration;
    // 半径从 5px 缓动扩散至 30px
    const radius = easeOut(elapsedRatio) * 25 + 5;
    // 透明度从 1 逐渐渐变为 0
    const opacity = easeOut(1 - elapsedRatio);

    const style = new Style({
      image: new CircleStyle({
        radius: radius,
        stroke: new Stroke({
          color: `rgba(255, 0, 0, ${opacity})`,
          width: 0.25 + opacity,
        }),
      }),
    });

    vectorContext.setStyle(style);
    vectorContext.drawGeometry(flashGeom);
    // 请求下一帧重绘继续推进动画
    map.render();
  }
}

3. 生产实战避坑与考量 (Production Pitfalls)

  • 挂载图层选择建议：动效监听建议挂载在底图图层（tileLayer.on('postrender')）或 map.on('postrender') 上，而不是动效要素所在的 vectorLayer。若挂载在 vectorLayer 上，可能因图层判定无自身数据变动而被跳过。
  • 多点并发动画合并调度：当成百上千个要素同时产生报警扩散波纹时，若为每个要素单独注册 postrender 监听器会导致几十甚至几百个监听函数在每帧重复执行。生产环境中应建立一个全局动画池，由单一 postrender 循环遍历绘制所有活动点位。
main.js 
import Feature from 'ol/Feature.js';
import Map from 'ol/Map.js';
import {unByKey} from 'ol/Observable.js';
import View from 'ol/View.js';
import {easeOut} from 'ol/easing.js';
import Point from 'ol/geom/Point.js';
import TileLayer from 'ol/layer/Tile.js';
import VectorLayer from 'ol/layer/Vector.js';
import {fromLonLat} from 'ol/proj.js';
import {getVectorContext} from 'ol/render.js';
import OSM from 'ol/source/OSM.js';
import VectorSource from 'ol/source/Vector.js';
import CircleStyle from 'ol/style/Circle.js';
import Stroke from 'ol/style/Stroke.js';
import Style from 'ol/style/Style.js';

const tileLayer = new TileLayer({
  source: new OSM({
    wrapX: false,
  }),
});

const source = new VectorSource({
  wrapX: false,
});
const vector = new VectorLayer({
  source: source,
});

const map = new Map({
  layers: [tileLayer, vector],
  target: 'map',
  view: new View({
    center: [0, 0],
    zoom: 1,
    multiWorld: true,
  }),
});

/**
 * Add a random point feature across the globe.
 * 在全球范围内随机生成并添加点要素。
 */
function addRandomFeature() {
  const x = Math.random() * 360 - 180;
  const y = Math.random() * 170 - 85;
  const geom = new Point(fromLonLat([x, y]));
  const feature = new Feature(geom);
  source.addFeature(feature);
}

const duration = 3000;

/**
 * Trigger a 3-second animated radar pulse/flash wave on the given feature.
 * 对指定要素触发为期 3 秒的雷达波纹扩散动效。
 * @param {Feature} feature Target feature to flash.
 */
function flash(feature) {
  const start = Date.now();
  const flashGeom = feature.getGeometry().clone();
  // Listen to tile layer's postrender event to draw directly on canvas
  // 监听底图图层的 postrender 渲染完成事件，直接向当前帧 Canvas 绘制过渡图形
  const listenerKey = tileLayer.on('postrender', animate);

  function animate(event) {
    const frameState = event.frameState;
    const elapsed = frameState.time - start;
    if (elapsed >= duration) {
      // Unsubscribe listener when animation completes
      // 动画超时，解绑注销事件，停止无谓重绘
      unByKey(listenerKey);
      return;
    }
    const vectorContext = getVectorContext(event);
    const elapsedRatio = elapsed / duration;
    // Radius expands from 5 to 30px with easeOut
    // 半径从 5px 随缓动函数扩展至 30px
    const radius = easeOut(elapsedRatio) * 25 + 5;
    const opacity = easeOut(1 - elapsedRatio);

    const style = new Style({
      image: new CircleStyle({
        radius: radius,
        stroke: new Stroke({
          color: 'rgba(255, 0, 0, ' + opacity + ')',
          width: 0.25 + opacity,
        }),
      }),
    });

    vectorContext.setStyle(style);
    vectorContext.drawGeometry(flashGeom);
    // Tell OpenLayers to continue postrender animation loop
    // 告知地图引擎继续请求下一渲染帧推进动画
    map.render();
  }
}

// Automatically trigger flash animation whenever a new feature is added
// 监听数据源 addfeature 事件，自动触发光晕动画
source.on('addfeature', function (e) {
  flash(e.feature);
});

// Periodically generate a random feature every 1 second
// 定时器：每秒自动生成一个新点要素
window.setInterval(addRandomFeature, 1000);
