Example of exporting a map as a PDF. / 使用 jsPDF 库将当前地图导出为高质量 PDF 文档的示例。

Example of exporting a map as a PDF using the jsPDF library.

本示例演示了如何使用 jsPDF 库将 OpenLayers 地图导出为标准矢量/光栅结合的 PDF 文件。支持从 A0 到 A5 的国际标准纸张尺寸选择，以及 72 DPI 至 300 DPI 的专业印刷分辨率设定。

📚 学习记录与技术深度解读

1. 核心架构与设计思想

  • 离屏高清多图层合流（Multi-Canvas Compositing）： 现代 OpenLayers 地图可能由多个图层叠加而成（如 OSM 瓦片 Canvas、矢量要素 Canvas、WebGL 纹理等），且各个 Canvas 可能具有独立的 CSS opacity 透明度与 CSS transform 变换矩阵。为了导出单张完整的图像，代码动态创建离屏 mapCanvas，通过遍历 .ol-layer canvas 节点并逐一提取 2D 仿射变换矩阵（setTransform），在内存中完整复现与屏幕视口 100% 一致的图层合成结果。
  • 动态分辨率倍率放缩（Resolution Upscaling）： 屏幕常规显示仅需 72~96 DPI，而印刷输出通常要求 150 DPI（公文报表）甚至 300 DPI（高精出版物）。本示例通过计算物理纸张尺寸（毫米）与目标 DPI 换算出对应的超大物理像素分辨率： $$\text{width (px)} = \text{round}\left(\frac{\text{dim[0] (mm)} \times \text{DPI}}{25.4}\right)$$ 然后动态临时修改地图的 map.setSize([width, height])，迫使 OpenLayers 在后台请求更密集的高分切片并重绘，完成后再无缝恢复原始屏幕尺寸。

2. 源码与关键 API 剖析

  • map.once('rendercomplete', callback)：单次渲染就绪监听器。在调用 map.setSize() 和 view.setResolution() 后，瓦片加载与渲染需要若干异步绘制帧，必须等待 rendercomplete 触发后才能提取 Canvas 像素。
  • matrix = transform.match(/^matrix\(([^\(]*)\)$/)[1]：正则提取 DOM 上的 CSS transform 矩阵参数，并利用 mapContext.setTransform.apply(mapContext, matrix) 将位移与缩放矩阵无损注入离屏 Canvas。
  • pdf.addImage(dataUrl, 'JPEG', 0, 0, dim[0], dim[1])：将离屏 Canvas 生成的图像数据嵌入 jsPDF 页面中，严格以毫米为单位映射到 A4/A3 纸张版面。

3. 生产实践踩坑与性能调优

  • CORS 跨域污染画布（Tainted Canvas）： 若地图底图瓦片未配置 crossOrigin: 'anonymous'，当调用 canvas.toDataURL() 时浏览器会抛出 SecurityError: Failed to execute 'toDataURL' on 'HTMLCanvasElement': Tainted canvases may not be exported。必须保证所有瓦片源服务器配置了 Access-Control-Allow-Origin: *。
  • 超大纸张（A0 / 300 DPI）内存崩溃： A0 尺寸（1189mm × 841mm）在 300 DPI 下对应的 Canvas 像素高达 $14043 \times 9933 \approx 1.39$ 亿像素，需消耗超过 550MB 内存，极易导致移动端或低配 PC 浏览器崩溃。生产环境建议设置合理的 DPI 上限，或在服务端使用 Headless Chrome / Node-Canvas 执行渲染。
main.js 
import {jsPDF} from 'jspdf';
import Map from 'ol/Map.js';
import View from 'ol/View.js';
import WKT from 'ol/format/WKT.js';
import TileLayer from 'ol/layer/Tile.js';
import VectorLayer from 'ol/layer/Vector.js';
import OSM from 'ol/source/OSM.js';
import VectorSource from 'ol/source/Vector.js';

// Base raster layer
// 基础栅格瓦片图层
const raster = new TileLayer({
  source: new OSM(),
});

// Create vector feature from WKT
// 从 WKT 字符串创建矢量要素
const format = new WKT();
const feature = format.readFeature(
  'POLYGON((10.689697265625 -25.0927734375, 34.595947265625 ' +
    '-20.1708984375, 38.814697265625 -35.6396484375, 13.502197265625 ' +
    '-39.1552734375, 10.689697265625 -25.0927734375))',
);
feature.getGeometry().transform('EPSG:4326', 'EPSG:3857');

const vector = new VectorLayer({
  source: new VectorSource({
    features: [feature],
  }),
  opacity: 0.5,
});

// Initialize map
// 初始化地图
const map = new Map({
  layers: [raster, vector],
  target: 'map',
  view: new View({
    center: [0, 0],
    zoom: 2,
  }),
});

// Paper dimensions in millimeters
// 标准打印纸张尺寸（单位：毫米）
const dims = {
  a0: [1189, 841],
  a1: [841, 594],
  a2: [594, 420],
  a3: [420, 297],
  a4: [297, 210],
  a5: [210, 148],
};

const exportButton = document.getElementById('export-pdf');

// Handle export PDF click
// 处理导出 PDF 点击事件
exportButton.addEventListener(
  'click',
  function () {
    exportButton.disabled = true;
    document.body.style.cursor = 'progress';

    const format = document.getElementById('format').value;
    const resolution = Number(document.getElementById('resolution').value);
    const dim = dims[format];
    const width = Math.round((dim[0] * resolution) / 25.4);
    const height = Math.round((dim[1] * resolution) / 25.4);
    const size = map.getSize();
    const viewResolution = map.getView().getResolution();

    // Listen for render completion to export image
    // 监听渲染完成事件以导出离屏合成图像
    map.once('rendercomplete', function () {
      const mapCanvas = document.createElement('canvas');
      mapCanvas.width = width;
      mapCanvas.height = height;
      const mapContext = mapCanvas.getContext('2d');
      Array.prototype.forEach.call(
        document.querySelectorAll('.ol-layer canvas'),
        function (canvas) {
          if (canvas.width > 0) {
            const opacity = canvas.parentNode.style.opacity;
            mapContext.globalAlpha = opacity === '' ? 1 : Number(opacity);
            const transform = canvas.style.transform;
            // Get the transform parameters from the style's transform matrix
            // 从 DOM 样式的变换矩阵中提取仿射参数
            const matrix = transform
              .match(/^matrix\(([^\(]*)\)$/)[1]
              .split(',')
              .map(Number);
            // Apply the transform to the export map context
            // 将仿射变换应用到导出画布上下文
            CanvasRenderingContext2D.prototype.setTransform.apply(
              mapContext,
              matrix,
            );
            mapContext.drawImage(canvas, 0, 0);
          }
        },
      );
      mapContext.globalAlpha = 1;
      mapContext.setTransform(1, 0, 0, 1, 0, 0);
      const pdf = new jsPDF('landscape', undefined, format);
      pdf.addImage(
        mapCanvas.toDataURL('image/jpeg'),
        'JPEG',
        0,
        0,
        dim[0],
        dim[1],
      );
      pdf.save('map.pdf');

      // Reset original map size
      // 恢复原始地图尺寸与分辨率
      map.setSize(size);
      map.getView().setResolution(viewResolution);
      exportButton.disabled = false;
      document.body.style.cursor = 'auto';
    });

    // Set print size and scale
    // 临时应用高分辨率打印尺寸与缩放比
    const printSize = [width, height];
    map.setSize(printSize);
    const scaling = Math.min(width / size[0], height / size[1]);
    map.getView().setResolution(viewResolution / scaling);
  },
  false,
);
