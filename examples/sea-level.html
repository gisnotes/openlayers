Render sea level at different elevations. / 演示如何使用 ol/source/Raster 对 MapTiler Terrain-RGB 地形切片进行像素级高程解算，动态渲染不同海平面淹没高度并进行实时交互。

This example uses a ol/source/Raster with MapTiler Terrain-RGB tiles to "flood" areas below the elevation shown on the sea level slider.

本示例演示了如何利用 ol/source/Raster 对 MapTiler Terrain-RGB 地形瓦片 执行客户端逐像素高程解码，并动态模拟海平面上升（Sea Level Rise）的淹没泛洪效果。

通过滑动条调节海平面相对上升高度（0 ~ 100 米），光栅数据源在后台 Web Worker 中实时将地表海拔低于该阈值的陆地区域渲染为半透明水体；同时监听光标移动，调用 raster.getData(coordinate) 实时提取并显示鼠标指针所在位置的亚米级真实地表海拔，并提供旧金山、纽约、孟买、上海等沿海重点城市的快速视角跳转。

--- ### 📚 学习记录与技术深度解读 (Learning Notes) #### 1. 核心架构与设计机制 (Core Architecture & Mechanics) - Mapbox / MapTiler Terrain-RGB 编码协议规范： Terrain-RGB 是目前数字高程切片（DEM Tiles）的行业通行标准，其高程（单位：米）解码公式为： $$\text{Elevation (m)} = -10000 + (R \times 256^2 + G \times 256 + B) \times 0.1$$ - 基础偏移量：$-10000\text{ m}$，以涵盖全球最深的海沟（如马里亚纳海沟 $-10984\text{ m}$ 附近）； - 垂直精度：$0.1\text{ m}$（10 厘米），足以支持微观城市洪涝与地形分析； - 动态范围：$256^3 \times 0.1 - 10000 \approx 1667721\text{ m}$，覆盖地球所有陆地山峰与深海。 - 地形源必须禁用双线性颜色插值 (`interpolate: false`)： 这是 Terrain-RGB 处理中最致命的工程避坑点！默认情况下浏览器和 WebGL 会对图像切片进行双线性插值（Bilinear Filter），这在普通视觉照片上能获得平滑过渡，但在以字节数值为语义的 Terrain-RGB 上，插值会导致 $R$、$G$、$B$ 通道独立混合，从而在切片边界或不同色块交界处产生极其怪异的几千米高程尖刺或深谷。在实例化 `ImageTile` 时必须显式声明 `interpolate: false`（强制采用最近邻采样 Nearest Neighbor）。

2. 关键代码解析与 API 模式 (Code Deep Dive & API Patterns) - 高性能 Web Worker 阈值泛洪着色算子：

function flood(pixels, data) {
  const pixel = pixels[0];
  if (pixel[3]) { // Alpha 通道存在有效数据
    const height = -10000 + (pixel[0] * 256 * 256 + pixel[1] * 256 + pixel[2]) * 0.1;
    if (height <= data.level) {
      // 低于设定水位线，涂装为浅蓝色水体
      pixel[0] = 134; pixel[1] = 203; pixel[2] = 249; pixel[3] = 255;
    } else {
      pixel[3] = 0; // 高于水位线完全透明，透出底图街道
    }
  }
  return pixel;
}

3. 工程避坑与高可用实践 (Production Pitfalls & Best Practices) - 地形切片最大层级与超视口拉伸：

Terrain-RGB 切片通常最大仅预生成至 z: 14，而城市街区矢量底图可达 z: 22。代码中为地形源配置了 maxZoom: 14，当用户继续放大时，OpenLayers 会自动将 14 级切片进行空间几何坐标拉伸放大展示，防止放大后高程图层突然空白消失。

  • 水体淹没与水动力拓扑连通性区别： 需注意本示例展示的是纯几何海拔阈值切分（Simple Inundation Model）。真实水动力学（如风暴潮、海平面上升）中，地势低洼但四周有防洪堤坝环抱的内陆盆地（如荷兰围海造田区）不会直接被海水淹没。若需高精度灾情评估，建议结合 region-growing（区域生长）算法，以入海口为种子点进行水系连通性阻隔计算。
main.js 
import Map from 'ol/Map.js';
import View from 'ol/View.js';
import ImageLayer from 'ol/layer/Image.js';
import TileLayer from 'ol/layer/Tile.js';
import {fromLonLat} from 'ol/proj.js';
import ImageTile from 'ol/source/ImageTile.js';
import RasterSource from 'ol/source/Raster.js';

/**
 * Flood operation decoding elevation and coloring pixels below water level.
 * 解码像素高程并将低于设定水位的陆地区域染成水体蓝色的泛洪运算函数
 * @param {Array<Array<number>>} pixels Input pixel arrays / 输入像素通道数组
 * @param {Object} data Parameters from beforeoperations / 外部传入参数对象
 * @return {Array<number>} Output pixel / 运算输出的像素
 */
function flood(pixels, data) {
  const pixel = pixels[0];
  if (pixel[3]) {
    // Mapbox/MapTiler Terrain-RGB decoding formula: height = -10000 + (R*256*256 + G*256 + B) * 0.1
    // Mapbox/MapTiler Terrain-RGB 标准解码公式
    const height =
      -10000 + (pixel[0] * 256 * 256 + pixel[1] * 256 + pixel[2]) * 0.1;
    if (height <= data.level) {
      // Inundated sea water color / 淹没水体赋浅蓝色
      pixel[0] = 134;
      pixel[1] = 203;
      pixel[2] = 249;
      pixel[3] = 255;
    } else {
      // Transparent to reveal base street map / 高于海平面区域完全透明透出街道底图
      pixel[3] = 0;
    }
  }
  return pixel;
}

const key = 'Get your own API key at https://www.maptiler.com/cloud/';
const attributions =
  '<a href="https://www.maptiler.com/copyright/" target="_blank">&copy; MapTiler</a> ' +
  '<a href="https://www.openstreetmap.org/copyright" target="_blank">&copy; OpenStreetMap contributors</a>';

// Terrain-RGB elevation tile source / Terrain-RGB 数字高程切片数据源
const elevation = new ImageTile({
  // The RGB values in the source collectively represent elevation.
  // Interpolation of individual colors would produce incorrect elevations and is disabled.
  // RGB 通道联合表示数值高程，通道独立平滑插值会产生严重高程突变，故强制禁用插值
  url:
    'https://api.maptiler.com/tiles/terrain-rgb-v2/{z}/{x}/{y}.webp?key=' + key,
  tileSize: 512,
  maxZoom: 14,
  interpolate: false,
});

// Raster source running flood calculation in Web Worker / 执行海平面泛洪运算的光栅数据源
const raster = new RasterSource({
  sources: [elevation],
  operation: flood,
});

// Initialize map with street basemap and sea level raster overlay / 初始化地图
const map = new Map({
  target: 'map',
  layers: [
    new TileLayer({
      source: new ImageTile({
        attributions: attributions,
        url:
          'https://api.maptiler.com/maps/streets-v2/{z}/{x}/{y}.png?key=' + key,
        tileSize: 512,
        maxZoom: 22,
      }),
    }),
    new ImageLayer({
      opacity: 0.6,
      source: raster,
    }),
  ],
  view: new View({
    center: fromLonLat([-122.3267, 37.8377]),
    zoom: 11,
  }),
});

// getData() returns the input source values at a coordinate; here the single
// input is the terrain-RGB source, so we decode its pixel to an elevation.
// getData() 返回指定坐标处的原始像素数组，从中解码并显示真实海拔
const elevationOutput = document.getElementById('elevation');
map.on(['pointermove', 'click'], function (event) {
  const pixels = raster.getData(event.coordinate);
  const pixel = pixels && pixels[0];
  if (!pixel || !pixel[3]) {
    elevationOutput.innerText = ' ';
    return;
  }
  const height =
    -10000 + (pixel[0] * 256 * 256 + pixel[1] * 256 + pixel[2]) * 0.1;
  elevationOutput.innerText = Math.round(height) + ' m';
});

// Sea level slider control / 海平面滑块控件联动
const control = document.getElementById('level');
const output = document.getElementById('output');
control.addEventListener('input', function () {
  output.innerText = control.value;
  raster.changed();
});
output.innerText = control.value;

// Pass slider value to worker before operations execute / 计算执行前将水位线数值传入 Worker
raster.on('beforeoperations', function (event) {
  event.data.level = control.value;
});

// Quick navigation buttons for coastal metropolitan areas / 沿海重点城市视角快速定位
const locations = document.getElementsByClassName('location');
for (let i = 0, ii = locations.length; i < ii; ++i) {
  locations[i].addEventListener('click', relocate);
}

/**
 * Relocate map view to selected city. / 将地图视角平移缩放至目标城市
 * @param {Event} event Click event / 点击事件
 */
function relocate(event) {
  const data = event.target.dataset;
  const view = map.getView();
  view.setCenter(fromLonLat(data.center.split(',').map(Number)));
  view.setZoom(Number(data.zoom));
}
