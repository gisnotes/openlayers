Example of a map with layer group. 本示例展示如何利用 ol/layer/Group 构建多层次的图层树状管理体系。示例将底图与若干 TileJSON 专题图层分层归类，并通过轻量级递归算法，自动为树形结构中的每个叶子图层与父图层组生成联动控件，实现图层组整体可见性开关、各图层独立不透明度（Opacity）动态调节与双向数据同步绑定。

📚 学习记录与技术深度解读 (Learning Notes)

1. 核心架构原理 (Core Architecture) 在复杂 GIS 项目中，图层动辄数十上百个，扁平化管理会导致图层顺序混乱、批量显隐难以控制。 - ol/layer/Group 复合设计模式（Composite Pattern）：LayerGroup 自身继承自 BaseLayer，因此它拥有与单一图层完全一致的 API（如 getVisible()、setVisible()、getOpacity()、setOpacity()）。当隐藏一个图层组时，其子图层数组中的所有图层在渲染流程中均会被整体跳过；当调节图层组的不透明度时，子图层的不透明度会与父级产生相乘效应：$\text{FinalOpacity} = \text{ParentOpacity} \times \text{ChildOpacity}$。 - 图层集合递归绑定（Recursive Tree Walk）：通过对 map.getLayerGroup() 执行深度优先遍历，若遇到 LayerGroup 实例则递归进入，自动为 DOM 树映射出唯一的层级 ID（如 #layer0, #layer10, #layer11），大幅简化动态图层目录（Layer Tree）的构建。

2. 关键实现与代码拆解 (Code Deep Dive) ```javascript // 1. 递归构建树级绑定 function setup(id, group) {

group.getLayers().forEach(function (layer, i) { const layerid = id + i; bindInputs(layerid, layer); // 若当前图层是图层组，则递归向下遍历子图层 if (layer instanceof LayerGroup) { setup(layerid, layer); } }); } // 从地图根图层组发起全局递归扫描 setup('#layer', map.getLayerGroup()); ```

3. 生产实战避坑与考量 (Production Pitfalls) - 父子透明度叠加：牢记 LayerGroup 的透明度会级联作用于所有子图层。若父级透明度设为 0.5，子级即使设为 1.0，屏幕最终呈现依然是 0.5。 - 动态增删图层监听：对于支持动态加载图层的业务系统，建议监听 group.getLayers().on('add' / 'remove')，保持 DOM 树节点与地图运行时内核完全同步。

main.js 
import Map from 'ol/Map.js';
import View from 'ol/View.js';
import LayerGroup from 'ol/layer/Group.js';
import TileLayer from 'ol/layer/Tile.js';
import {fromLonLat} from 'ol/proj.js';
import OSM from 'ol/source/OSM.js';
import TileJSON from 'ol/source/TileJSON.js';

const key =
  'pk.eyJ1IjoiYWhvY2V2YXIiLCJhIjoiY2t0cGdwMHVnMGdlbzMxbDhwazBic2xrNSJ9.WbcTL9uj8JPAsnT9mgb7oQ';

// Initialize map with individual layer and a nested LayerGroup
// 初始化包含单图层与嵌套图层组（LayerGroup）的地图实例
const map = new Map({
  layers: [
    new TileLayer({
      source: new OSM(),
    }),
    new LayerGroup({
      layers: [
        new TileLayer({
          source: new TileJSON({
            url:
              'https://api.tiles.mapbox.com/v4/mapbox.20110804-hoa-foodinsecurity-3month.json?secure&access_token=' +
              key,
            crossOrigin: 'anonymous',
          }),
        }),
        new TileLayer({
          source: new TileJSON({
            url:
              'https://api.tiles.mapbox.com/v4/mapbox.world-borders-light.json?secure&access_token=' +
              key,
            crossOrigin: 'anonymous',
          }),
        }),
      ],
    }),
  ],
  target: 'map',
  view: new View({
    center: fromLonLat([37.4057, 8.81566]),
    zoom: 4,
  }),
});

/**
 * Bind input elements (checkbox & slider) to a layer's visibility and opacity.
 * 将 DOM 表单控件（复选框与滑动条）与图层的可见性及透明度进行双向绑定。
 * @param {string} layerid DOM element selector prefix.
 * @param {import('ol/layer/Base.js').default} layer Target layer.
 */
function bindInputs(layerid, layer) {
  const visibilityInput = document.querySelector(layerid + ' input.visible');
  visibilityInput.addEventListener('change', function () {
    layer.setVisible(this.checked);
  });
  visibilityInput.checked = layer.getVisible();

  const opacityInput = document.querySelector(layerid + ' input.opacity');
  opacityInput.addEventListener('input', function () {
    layer.setOpacity(parseFloat(this.value));
  });
  opacityInput.value = String(layer.getOpacity());
}

/**
 * Recursively traverse layer groups and bind controls to each layer.
 * 递归遍历图层树，为每个叶子图层与父图层组动态绑定 UI 控件。
 * @param {string} id ID prefix.
 * @param {LayerGroup} group Layer group.
 */
function setup(id, group) {
  group.getLayers().forEach(function (layer, i) {
    const layerid = id + i;
    bindInputs(layerid, layer);
    if (layer instanceof LayerGroup) {
      setup(layerid, layer);
    }
  });
}
setup('#layer', map.getLayerGroup());

// Toggle visibility of property fieldset when clicking layer titles
// 点击图层标题展开/折叠对应的属性调节表单
document.querySelectorAll('#layertree li > span').forEach(function (element) {
  element.addEventListener('click', function () {
    this.parentNode.querySelector('fieldset').style.display =
      this.parentNode.querySelector('fieldset').style.display === 'none'
        ? ''
        : 'none';
  });
  element.parentNode.querySelector('fieldset').style.display = 'none';
});
