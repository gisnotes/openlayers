Shaded Relief (with WebGL) / WebGL 动态山体阴影（地貌日照晕渲与邻域采样）

webgl19 shadedrelief2

x
°
°

Calculate shaded relief from elevation data. / 利用 WebGL 邻域像元采样算子在片元着色器中实时计算坡度与坡向，实现可动态调整光源方向的山体日照晕渲效果。

For the shaded relief, a single tiled source of elevation data is used as input. The shaded relief is calculated by the layer's style with a color expression. The style variables are updated when the user drags one of the sliders. The band operator is used to sample data from neighboring pixels for calculating slope and aspect, which is done with the ['band', bandIndex, xOffset, yOffset] syntax.

本示例利用单一的瓦片高程数据源作为输入生成地表动态山体阴影（Shaded Relief / 日照晕渲）。山体阴影通过 WebGL 瓦片图层 style 中的 color 声明式表达式在 GPU 片元着色器中实时解算。拖动下方滑块可动态更新样式变量。着色器中使用具备像元偏移能力的 band 算子对当前像素周围的邻域像元进行采样（语法为 ['band', bandIndex, xOffset, yOffset]），从而精确解算出各点的坡度（Slope）与坡向（Aspect）。

📚 学习记录与技术深度解读 (Learning Notes)

1. 邻域像元偏移采样语法：['band', bandIndex, xOffset, yOffset]

  • 突破单像素局限：常规的片元着色仅能读取当前坐标对应的一个像素值。而在空间卷积滤波、边缘提取（Sobel）及地形表面分析中，必须获取相邻栅格数据。
  • 硬件级纹素偏移（Texel Offset）：OpenLayers WebGL 瓦片引擎通过 ['band', n, dx, dy] 允许传入整数像素偏移量（例如 [-1, 0] 代表西侧相邻像素，[1, 0] 代表东侧相邻像素）。底层着色器自动换算纹理坐标偏移，实现极低开销的邻域采样。

2. GPU 实时解算中心差分、坡度、坡向与 Lambert 漫反射光照

  • 高程梯度中心差分： $$\frac{\partial z}{\partial x} = \frac{Z_{x+1} - Z_{x-1}}{2 \times \text{resolution}}$$ $$\frac{\partial z}{\partial y} = \frac{Z_{y+1} - Z_{y-1}}{2 \times \text{resolution}}$$
  • 坡度（Slope）与坡向（Aspect）： $$\text{Slope} = \arctan\left(\sqrt{(\partial z / \partial x)^2 + (\partial z / \partial y)^2}\right)$$ $$\text{Aspect} = \arctan\left(-\frac{\partial z}{\partial x}, \frac{\partial z}{\partial y}\right)$$
  • 太阳光入射余弦角（Cosine Incidence）： $$\cos(\theta) = \sin(\text{SunEl}) \cdot \cos(\text{Slope}) + \cos(\text{SunEl}) \cdot \sin(\text{Slope}) \cdot \cos(\text{SunAz} - \text{Aspect})$$
  • 全动态可交互性：传统 GIS 晕渲是预切的静态死图，而此方案在客户端可任意拖动太阳高度角（sunEl）与太阳方位角（sunAz），实时模拟晨昏、正午与季节太阳照射的光影变幻，视觉冲击力极强。
main.js 
import Map from 'ol/Map.js';
import View from 'ol/View.js';
import TileLayer from 'ol/layer/WebGLTile.js';
import ImageTile from 'ol/source/ImageTile.js';
import OSM from 'ol/source/OSM.js';

const variables = {};

// Method used to extract elevations from the DEM.
// In this case the format used is Terrarium: (red * 256 + green + blue / 256) - 32768
// Other frequently used methods include the Mapbox format: (red * 256 * 256 + green * 256 + blue) * 0.1 - 10000
// 从 DEM 瓦片中提取真实海拔高程的方法。
// 本例使用的瓦片编码格式为 Mapzen Terrarium：red * 256 + green + blue / 256 - 32768
function elevation(xOffset, yOffset) {
  const red = ['band', 1, xOffset, yOffset];
  const green = ['band', 2, xOffset, yOffset];
  const blue = ['band', 3, xOffset, yOffset];

  // Band math operates on normalized values from 0-1, so we scale by 255
  // 波段数学运算使用 0-1 范围的归一化像素值，因此乘以 255 对齐字节分量
  return [
    '+',
    ['*', 255 * 256, red],
    ['*', 255, green],
    ['*', 255 / 256, blue],
    -32768,
  ];
}

// Generates shaded relief image given elevation data. Uses 3x3 neighborhood for determining slope and aspect.
// 根据高程数据在着色器中生成山体日照阴影。利用 3x3 邻域中心差分计算地表坡度（Slope）与坡向（Aspect）。
const dp = ['*', 2, ['resolution']];
const z0x = ['*', ['var', 'vert'], elevation(-1, 0)];
const z1x = ['*', ['var', 'vert'], elevation(1, 0)];
const dzdx = ['/', ['-', z1x, z0x], dp];
const z0y = ['*', ['var', 'vert'], elevation(0, -1)];
const z1y = ['*', ['var', 'vert'], elevation(0, 1)];
const dzdy = ['/', ['-', z1y, z0y], dp];
const slope = ['atan', ['sqrt', ['+', ['^', dzdx, 2], ['^', dzdy, 2]]]];
const aspect = ['clamp', ['atan', ['-', 0, dzdx], dzdy], -Math.PI, Math.PI];
const sunEl = ['*', Math.PI / 180, ['var', 'sunEl']];
const sunAz = ['*', Math.PI / 180, ['var', 'sunAz']];

// Lambertian cosine incidence angle / 漫反射光照入射角余弦值计算
const cosIncidence = [
  '+',
  ['*', ['sin', sunEl], ['cos', slope]],
  ['*', ['cos', sunEl], ['sin', slope], ['cos', ['-', sunAz, aspect]]],
];
const scaled = ['*', 255, cosIncidence];

const shadedRelief = new TileLayer({
  opacity: 0.3,
  source: new ImageTile({
    url: 'https://s3.amazonaws.com/elevation-tiles-prod/terrarium/{z}/{x}/{y}.png',
    maxZoom: 15,
    attributions:
      '<a href="https://github.com/tilezen/joerd/blob/master/docs/attribution.md" target="_blank">Data sources and attribution</a>',
  }),
  style: {
    variables: variables,
    color: ['color', scaled, scaled, scaled],
  },
});

// Update style variables and reflect them in UI text spans
// 动态更新样式变量并同步界面文本显示
const controlIds = ['vert', 'sunEl', 'sunAz'];
controlIds.forEach(function (id) {
  const control = document.getElementById(id);
  const output = document.getElementById(id + 'Out');
  function value() {
    return parseFloat(control.value);
  }
  variables[id] = value();
  function update() {
    output.innerText = control.value;
    variables[id] = value();
    shadedRelief.updateStyleVariables(variables);
  }
  control.addEventListener('input', update);
  update();
});

const map = new Map({
  target: 'map',
  layers: [
    new TileLayer({
      source: new OSM(),
    }),
    shadedRelief,
  ],
  view: new View({
    center: [15016024, -4076722],
    zoom: 11,
  }),
});
