Using WebGL to filter large quantities of features. / 使用 WebGL 动态过滤海量点要素并结合着色器实现呼吸脉冲动画。

This example shows how to use ol/layer/WebGLVector with a literal style to dynamically filter a large amount of point geometries. The above map is based on a dataset from the NASA containing 45k recorded meteorite landing sites. Each meteorite is marked by a circle on the map (the bigger the circle, the heavier the object). A pulse effect has been added, which is slightly offset by the year of the impact.

Adjusting the sliders causes the objects outside of the date range to be filtered out of the map. This is done by mutating the variables used by the style object provided to the WebGL layer. Also note that the last snippet of code is necessary to make sure the map refreshes itself every frame.

本示例演示如何使用 ol/layer/WebGLVector 搭配声明式字面量样式（Literal Style），动态过滤海量点几何要素。底图数据源来自 NASA 记录的全球约 45,000 处陨石撞击点。每个撞击点在地图上以圆点标绘（圆点越大代表陨石质量越重），并根据撞击发生年份加入了带有微小相位偏移的脉冲动态呼吸光晕效果。

拖动下方年份滑块可动态过滤超出时间范围的要素。该过程通过修改传递给 WebGL 矢量图层 style 对象的样式变量（Variables）完成。此外，文末使用 requestAnimationFrame 触发持续重绘，以驱动 GPU 片元着色器实时计算逐帧脉冲动画。

📚 学习记录与技术深度解读 (Learning Notes)

1. GPU 实例绘制（Instanced Rendering）与海量点承载

  • 传统 Canvas 瓶颈：在普通 Canvas 2D 渲染下，绘制 4.5 万个具有动态缩放、透明度和渐变色的要素会导致极严重的 CPU 循环绘制卡顿（帧率跌至 10 FPS 以下）。
  • WebGL 批量提交：ol/layer/WebGLVector 将所有点的坐标、属性（质量 mass、年份 year）打包为显存顶点缓冲区（VBO），通过一次 Draw Call 驱动 GPU 进行硬件级实例绘制，实现 60 FPS 满帧渲染。

2. 着色器内置时间运算符 ['time'] 与呼吸脉冲

  • 时间驱动插值：样式表达式中使用了 ['time'] 运算符，该算子读取当前动画帧时间戳（以秒为单位）。
  • 连续波动函数： animRatio = ( (time + yearOffset) % period ) / period ^ 0.5 利用求模与指数运算构造出周期为 12 秒的非线性脉冲曲线。各点根据自身的撞击年份（year）产生错落有致的起伏，无需在 JS 层面逐点计算属性更新。

3. 毫秒级属性过滤与命中探测优化

  • 片元级条件过滤：filter: ['between', ['get', 'year'], ['var', 'minYear'], ['var', 'maxYear']]。滑块变动时调用 pointsLayer.updateStyleVariables()，GPU 片元着色器直接根据 Uniform 变量丢弃非区间点位，无任何要素重建或垃圾回收停顿。
  • disableHitDetection: true 性能保护：在超大规模数据集下，如果不需要鼠标点击/悬停拾取要素，显式关闭命中探测可完全避免在离屏 Hit-Buffer 上的额外渲染开销。
main.js 
import Feature from 'ol/Feature.js';
import Map from 'ol/Map.js';
import View from 'ol/View.js';
import Point from 'ol/geom/Point.js';
import TileLayer from 'ol/layer/Tile.js';
import WebGLVectorLayer from 'ol/layer/WebGLVector.js';
import {fromLonLat} from 'ol/proj.js';
import StadiaMaps from 'ol/source/StadiaMaps.js';
import Vector from 'ol/source/Vector.js';

const vectorSource = new Vector({
  attributions: 'NASA',
});

const oldColor = 'rgba(242,56,22,0.61)';
const newColor = '#ffe52c';
const period = 12; // Animation period in seconds / 动画周期（秒）
// Construct periodic pulse ratio using shader math operators
// 使用着色器数学运算符构造周期性呼吸脉冲比率
const animRatio = [
  '^',
  [
    '/',
    [
      '%',
      [
        '+',
        ['time'],
        ['interpolate', ['linear'], ['get', 'year'], 1850, 0, 2015, period],
      ],
      period,
    ],
    period,
  ],
  0.5,
];

// WebGL style expressions mapping mass and animation ratio to radius, color, and opacity
// WebGL 样式表达式：将质量和动画比率映射到半径、填充色和透明度
const style = {
  'circle-radius': [
    '*',
    ['interpolate', ['linear'], ['get', 'mass'], 0, 4, 200000, 13],
    ['-', 1.75, ['*', animRatio, 0.75]],
  ],
  'circle-fill-color': [
    'interpolate',
    ['linear'],
    animRatio,
    0,
    newColor,
    1,
    oldColor,
  ],
  'circle-opacity': ['-', 1.0, ['*', animRatio, 0.75]],
};

// Handle input slider values and events / 处理滑块输入值与事件
const minYearInput = document.getElementById('min-year');
const maxYearInput = document.getElementById('max-year');

function updateStatusText() {
  const div = document.getElementById('status');
  div.querySelector('span.min-year').textContent = minYearInput.value;
  div.querySelector('span.max-year').textContent = maxYearInput.value;
}

const pointsLayer = new WebGLVectorLayer({
  variables: {
    minYear: parseInt(minYearInput.value, 10),
    maxYear: parseInt(maxYearInput.value, 10),
  },
  style: [
    {
      style,
      filter: [
        'between',
        ['get', 'year'],
        ['var', 'minYear'],
        ['var', 'maxYear'],
      ],
    },
  ],
  source: vectorSource,
  disableHitDetection: true, // Disable hit detection for optimal performance with 45k points / 禁用拾取探测以提升海量点性能
});

minYearInput.addEventListener('input', function () {
  pointsLayer.updateStyleVariables({
    minYear: parseInt(minYearInput.value, 10),
  });
  updateStatusText();
});
maxYearInput.addEventListener('input', function () {
  pointsLayer.updateStyleVariables({
    maxYear: parseInt(maxYearInput.value, 10),
  });
  updateStatusText();
});
updateStatusText();

// Load CSV data / 异步加载陨石撞击 CSV 数据
const client = new XMLHttpRequest();
client.open('GET', 'data/csv/meteorite_landings.csv');
client.onload = function () {
  const csv = client.responseText;
  const features = [];

  let prevIndex = csv.indexOf('\n') + 1; // Scan past header line / 跳过表头首行

  let curIndex;
  while ((curIndex = csv.indexOf('\n', prevIndex)) != -1) {
    const line = csv.substr(prevIndex, curIndex - prevIndex).split(',');
    prevIndex = curIndex + 1;

    const coords = fromLonLat([parseFloat(line[4]), parseFloat(line[3])]);
    if (isNaN(coords[0]) || isNaN(coords[1])) {
      // Guard against bad or missing data / 过滤无效坐标数据
      continue;
    }

    features.push(
      new Feature({
        mass: parseFloat(line[1]) || 0,
        year: parseInt(line[2], 10) || 0,
        geometry: new Point(coords),
      }),
    );
  }

  vectorSource.addFeatures(features);
};
client.send();

const map = new Map({
  layers: [
    new TileLayer({
      source: new StadiaMaps({
        layer: 'stamen_toner',
      }),
    }),
    pointsLayer,
  ],
  target: document.getElementById('map'),
  view: new View({
    center: [0, 0],
    zoom: 2,
  }),
});

// Animate map continuously every frame / 逐帧驱动地图重绘以展示着色器脉冲动画
function animate() {
  map.render();
  window.requestAnimationFrame(animate);
}
animate();
