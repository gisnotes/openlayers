Custom Interactions / 自定义交互开发

drag5 feature12 vector87 editing2 custom7 interaction6

Example of a custom interaction. / 演示通过继承 ol/interaction/Pointer 开发自定义拖拽手势交互。

This example demonstrates creating a custom interaction by subclassing ol/interaction/Pointer. Note that the built in interaction ol/interaction/Translate might be a better option for moving features.

本示例演示如何通过继承 ol/interaction/Pointer 类来编写自定义的指针手势交互。 （注：在实际生产项目中，如果仅仅是为了移动平移要素，使用官方内置的 ol/interaction/Translate 交互通常是更优、更高效的选择；本示例核心在于展示底层手势与交互扩展的标准骨架与状态机模式）。

📚 学习记录与技术深度解读 (Learning Notes)

1. 继承 ol/interaction/Pointer 的四大手势生命周期函数

  • handleDownEvent(evt)：
    • 当鼠标左键或触控手指按下时触发。
    • 核心返回值机制：返回 true 表示当前交互成功“捕获并认领”了该事件序列（例如命中要素），系统将自动锁定当前手势并将接下来的 Drag 事件全部派发给它；返回 false 则放弃接管，事件透传给底层的 DragPan 实现地图拖拽。
  • handleDragEvent(evt)：
    • 仅在 handleDownEvent 返回 true 且光标移动时被持续调用。
    • 通过当前坐标减去上一次坐标计算增量位移（$\Delta x, \Delta y$），调用 geometry.translate(deltaX, deltaY) 实时移动要素。
  • handleMoveEvent(evt)：
    • 鼠标未按下时的自由悬停移动（Hover）。
    • 调用 map.forEachFeatureAtPixel 探测目标，若悬停在可拖拽要素上方，动态将容器鼠标光标切换为手型（pointer），提供直观交互预备反馈。
  • handleUpEvent(evt)：
    • 鼠标抬起或手指离开屏幕时触发。
    • 清理内部引用的 coordinate_ 与 feature_，返回 false 宣告本次拖拽生命周期正式结束并释放事件锁定。

2. 自定义交互的适用场景

  • 当面对复杂工业业务逻辑（如要素引出连接线、三维倾斜视角旋转手柄、自定义角度旋转尺、量测标注二次微调）时，基于 PointerInteraction 派生是扩展 OpenLayers 能力最干净、最彻底的标准途径。
main.js 
import Feature from 'ol/Feature.js';
import Map from 'ol/Map.js';
import View from 'ol/View.js';
import LineString from 'ol/geom/LineString.js';
import Point from 'ol/geom/Point.js';
import Polygon from 'ol/geom/Polygon.js';
import PointerInteraction from 'ol/interaction/Pointer.js';
import {defaults as defaultInteractions} from 'ol/interaction/defaults.js';
import TileLayer from 'ol/layer/Tile.js';
import VectorLayer from 'ol/layer/Vector.js';
import OGCMapTile from 'ol/source/OGCMapTile.js';
import VectorSource from 'ol/source/Vector.js';

// Custom Drag interaction subclassing PointerInteraction
// 通过继承 PointerInteraction 自定义实现的要素平移拖拽交互类
class Drag extends PointerInteraction {
  constructor() {
    super({
      handleDownEvent: handleDownEvent,
      handleDragEvent: handleDragEvent,
      handleMoveEvent: handleMoveEvent,
      handleUpEvent: handleUpEvent,
    });

    /**
     * @type {import('ol/coordinate.js').Coordinate}
     * @private
     */
    this.coordinate_ = null;

    /**
     * @type {string|undefined}
     * @private
     */
    this.cursor_ = 'pointer';

    /**
     * @type {Feature}
     * @private
     */
    this.feature_ = null;

    /**
     * @type {string|undefined}
     * @private
     */
    this.previousCursor_ = undefined;
  }
}

/**
 * Handle pointer down event. Return true to claim the drag sequence.
 * 处理指针按下事件。返回 true 声明接管后续的拖拽事件序列。
 * @param {import('ol/MapBrowserEvent.js').default} evt Map browser event.
 * @return {boolean} `true` to start the drag sequence.
 */
function handleDownEvent(evt) {
  const map = evt.map;

  const feature = map.forEachFeatureAtPixel(evt.pixel, function (feature) {
    return feature;
  });

  if (feature) {
    this.coordinate_ = evt.coordinate;
    this.feature_ = feature;
  }

  return !!feature;
}

/**
 * Handle pointer drag event: calculate coordinate delta and translate the feature.
 * 处理指针拖拽事件：计算坐标偏移量并实时平移要素几何。
 * @param {import('ol/MapBrowserEvent.js').default} evt Map browser event.
 */
function handleDragEvent(evt) {
  const deltaX = evt.coordinate[0] - this.coordinate_[0];
  const deltaY = evt.coordinate[1] - this.coordinate_[1];

  const geometry = this.feature_.getGeometry();
  geometry.translate(deltaX, deltaY);

  this.coordinate_[0] = evt.coordinate[0];
  this.coordinate_[1] = evt.coordinate[1];
}

/**
 * Handle pointer move event: update cursor style on hover.
 * 处理指针悬停移动事件：鼠标悬停在要素上方时动态更新光标样式为手型。
 * @param {import('ol/MapBrowserEvent.js').default} evt Event.
 */
function handleMoveEvent(evt) {
  if (this.cursor_) {
    const map = evt.map;
    const feature = map.forEachFeatureAtPixel(evt.pixel, function (feature) {
      return feature;
    });
    const element = evt.map.getTargetElement();
    if (feature) {
      if (element.style.cursor != this.cursor_) {
        this.previousCursor_ = element.style.cursor;
        element.style.cursor = this.cursor_;
      }
    } else if (this.previousCursor_ !== undefined) {
      element.style.cursor = this.previousCursor_;
      this.previousCursor_ = undefined;
    }
  }
}

/**
 * Handle pointer up event: finish the drag sequence and release references.
 * 处理指针抬起事件：结束当前拖拽手势生命周期并释放引用。
 * @return {boolean} `false` to stop the drag sequence.
 */
function handleUpEvent() {
  this.coordinate_ = null;
  this.feature_ = null;
  return false;
}

const pointFeature = new Feature(new Point([0, 0]));

const lineFeature = new Feature(
  new LineString([
    [-1e7, 1e6],
    [-1e6, 3e6],
  ]),
);

const polygonFeature = new Feature(
  new Polygon([
    [
      [-3e6, -1e6],
      [-3e6, 1e6],
      [-1e6, 1e6],
      [-1e6, -1e6],
      [-3e6, -1e6],
    ],
  ]),
);

const map = new Map({
  interactions: defaultInteractions().extend([new Drag()]),
  layers: [
    new TileLayer({
      source: new OGCMapTile({
        url: 'https://maps.gnosis.earth/ogcapi/collections/NaturalEarth:raster:HYP_HR_SR_OB_DR/map/tiles/WebMercatorQuad',
        crossOrigin: '',
      }),
    }),
    new VectorLayer({
      source: new VectorSource({
        features: [pointFeature, lineFeature, polygonFeature],
      }),
      style: {
        'icon-src': 'https://cdn.jsdelivr.net/gh/gisnotes/openlayers@gh-pages/examples/data/icon.png',
        'icon-opacity': 0.95,
        'icon-anchor': [0.5, 46],
        'icon-anchor-x-units': 'fraction',
        'icon-anchor-y-units': 'pixels',
        'stroke-width': 3,
        'stroke-color': [255, 0, 0, 1],
        'fill-color': [0, 0, 255, 0.6],
      },
    }),
  ],
  target: 'map',
  view: new View({
    center: [0, 0],
    zoom: 2,
  }),
});
