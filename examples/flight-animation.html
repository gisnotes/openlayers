Flight Animation / 大圆航线生成与全球航班飞行轨迹动效

animation5 vector87 feature12 flights1 arc1

Demonstrates how to animate flights with ´postrender´. / 演示如何使用 arc.js 生成大圆弧线（Great Circle）并利用 postrender 驱动多路航班飞行轨迹动态延展。

This example shows how to use postrender and vectorContext to animate flights. A great circle arc between two airports is calculated using arc.js and then the flight paths are animated with postrender. The flight data is provided by OpenFlights (a simplified data set from the Mapbox.js documentation is used).

本示例展示如何利用 postrender 事件帧回调与 VectorContext 矢量管线呈现震撼的全球航班飞行轨迹动效。借助第三方数学库 arc.js 精确解算两座机场之间在球面几何上的大圆航线（Great Circle Arc），自动切分跨越国际日期变更线（$\pm 180^\circ$ 子午线）的断裂分段，并利用时间驱动的动态折线切片，配合多世界无缝平移（World Wrapping Offset），呈现数百条航线依次腾飞、平滑生长的航空大屏视效。航线数据来源于 OpenFlights。

📚 学习记录与技术深度解读 (Learning Notes)

1. 核心架构原理 (Core Architecture)

在地球曲面上，两点间的最短距离是球面大圆劣弧（Great Circle），而在墨卡托平面地图中大圆航线呈现为弯曲拱形曲线：

  • 大圆航线离散化与跨日界线反子午线切分：arc.GreatCircle 生成包含 100 个插值控制点的空间弧线。若航线跨越了 $\pm 180^\circ$ 反子午线，arc.js 会自动切分为两段独立的几何折线（LineString），避免在地图东西边缘横跨整幅画卷产生混乱的贯穿拉线。
  • 动态折线子切片截取 (coords.slice(0, maxIndex))：通过当前帧真实流逝时间计算已飞过的离散点索引： $$\text{maxIndex} = \min(\text{elapsedTime} \times \text{pointsPerMs}, \text{coords.length})$$ 构造临时截断折线 new LineString(coords.slice(0, maxIndex)) 并通过 vectorContext.drawGeometry() 瞬态上屏，逼真模拟出航迹由起点向终点持续“生长延伸”的粒子流光效果。
  • 跨世界连续平铺（World Wrapping）补偿：当用户横向无级拖拽地球时，视口中心可能位于第 $N$ 个复制世界。示例通过： $$\text{worldWidth} = \text{getWidth}(\text{projectionExtent}), \quad \text{offset} = \left\lfloor \frac{\text{center}_x}{\text{worldWidth}} \right\rfloor$$ 在主世界与相邻副本世界中连续调用 currentLine.translate(worldWidth, 0) 绘制两遍，确保环球漫游时航线无论滑到哪里都毫无接缝断层。

2. 关键实现与代码拆解 (Code Deep Dive)

function animateFlights(event) {
  const vectorContext = getVectorContext(event);
  const frameState = event.frameState;
  vectorContext.setStyle(style);

  const features = flightsSource.getFeatures();
  for (let i = 0; i < features.length; i++) {
    const feature = features[i];
    if (!feature.get('finished')) {
      const coords = feature.getGeometry().getCoordinates();
      const elapsedTime = frameState.time - feature.get('start');
      if (elapsedTime >= 0) {
        const elapsedPoints = elapsedTime * pointsPerMs;
        if (elapsedPoints >= coords.length) {
          feature.set('finished', true); // 飞行结束，移交常规图层样式渲染
        }
        // 截取当前飞行阶段的航迹折线
        const maxIndex = Math.min(elapsedPoints, coords.length);
        const currentLine = new LineString(coords.slice(0, maxIndex));

        // 多世界坐标平移与双重绘制补偿
        const worldWidth = getWidth(map.getView().getProjection().getExtent());
        const offset = Math.floor(map.getView().getCenter()[0] / worldWidth);
        currentLine.translate(offset * worldWidth, 0);
        vectorContext.drawGeometry(currentLine);
        currentLine.translate(worldWidth, 0);
        vectorContext.drawGeometry(currentLine);
      }
    }
  }
  // 持续驱动下一帧重绘
  map.render();
}

3. 生产实战避坑与考量 (Production Pitfalls)

  • 动画结束状态平滑移交：在航迹完全长成后，设置 feature.set('finished', true)，并将图层样式配置为 style: feature => feature.get('finished') ? style : null。此时要素平滑移交由矢量图层缓存机制进行常规渲染，不再占用每帧动态切片与临时对象创建，大幅降低稳定状态下的 CPU 开销。
  • 大量几何对象创建与垃圾回收（GC）优化：每帧 new LineString(slice) 会在高频循环中产生瞬态对象。对于超大规模航线网（万级航班），应复用固定的坐标缓冲区，或采用 WebGL 顶点着色器时序属性控制流线动效。
main.js 
import Feature from 'ol/Feature.js';
import Map from 'ol/Map.js';
import View from 'ol/View.js';
import {getWidth} from 'ol/extent.js';
import LineString from 'ol/geom/LineString.js';
import TileLayer from 'ol/layer/Tile.js';
import VectorLayer from 'ol/layer/Vector.js';
import {getVectorContext} from 'ol/render.js';
import StadiaMaps from 'ol/source/StadiaMaps.js';
import VectorSource from 'ol/source/Vector.js';
import Stroke from 'ol/style/Stroke.js';
import Style from 'ol/style/Style.js';

const tileLayer = new TileLayer({
  source: new StadiaMaps({
    layer: 'stamen_toner',
  }),
});

const map = new Map({
  layers: [tileLayer],
  target: 'map',
  view: new View({
    center: [-11000000, 4600000],
    zoom: 2,
  }),
});

const style = new Style({
  stroke: new Stroke({
    color: '#EAE911',
    width: 2,
  }),
});

const flightsSource = new VectorSource({
  attributions:
    'Flight data by ' +
    '<a href="https://openflights.org/data.html">OpenFlights</a>,',
  loader: function () {
    const url = 'data/openflights/flights.json';
    fetch(url)
      .then(function (response) {
        return response.json();
      })
      .then(function (json) {
        const flightsData = json.flights;
        for (let i = 0; i < flightsData.length; i++) {
          const flight = flightsData[i];
          const from = flight[0];
          const to = flight[1];

          // create an arc circle between the two locations
          // 利用 arc.js 在起点与终点之间计算球面大圆弧线
          const arcGenerator = new arc.GreatCircle(
            {x: from[1], y: from[0]},
            {x: to[1], y: to[0]},
          );

          const arcLine = arcGenerator.Arc(100, {offset: 10});
          // paths which cross the -180°/+180° meridian are split
          // into two sections which will be animated sequentially
          // 跨越 180° 反子午线的路径被自动切分为两个几何分段，按时序依次呈现动画
          const features = [];
          arcLine.geometries.forEach(function (geometry) {
            const line = new LineString(geometry.coords);
            line.transform('EPSG:4326', 'EPSG:3857');

            features.push(
              new Feature({
                geometry: line,
                finished: false,
              }),
            );
          });
          // add the features with a delay so that the animation
          // for all features does not start at the same time
          // 增加阶梯延迟，使航线错峰腾飞
          addLater(features, i * 50);
        }
        tileLayer.on('postrender', animateFlights);
      });
  },
});

const flightsLayer = new VectorLayer({
  source: flightsSource,
  style: function (feature) {
    // if the animation is still active for a feature, do not
    // render the feature with the layer style
    // 若要素正在进行动态飞行，图层不直接绘制；完全飞完后移交常规样式渲染
    if (feature.get('finished')) {
      return style;
    }
    return null;
  },
});

map.addLayer(flightsLayer);

const pointsPerMs = 0.02;

/**
 * Postrender animation handler: draws growing flight lines with vectorContext.
 * 渲染后帧回调：使用 vectorContext 瞬态绘制逐步生长的飞行航迹。
 * @param {import('ol/render/Event.js').default} event Render event.
 */
function animateFlights(event) {
  const vectorContext = getVectorContext(event);
  const frameState = event.frameState;
  vectorContext.setStyle(style);

  const features = flightsSource.getFeatures();
  for (let i = 0; i < features.length; i++) {
    const feature = features[i];
    if (!feature.get('finished')) {
      // only draw the lines for which the animation has not finished yet
      // 仅对尚未完成飞行的航线进行帧动画绘制
      const coords = feature.getGeometry().getCoordinates();
      const elapsedTime = frameState.time - feature.get('start');
      if (elapsedTime >= 0) {
        const elapsedPoints = elapsedTime * pointsPerMs;

        if (elapsedPoints >= coords.length) {
          feature.set('finished', true);
        }

        const maxIndex = Math.min(elapsedPoints, coords.length);
        const currentLine = new LineString(coords.slice(0, maxIndex));

        // animation is needed in the current and nearest adjacent wrapped world
        // 多世界坐标平移补偿：计算当前视口中心所在的平铺世界偏移
        const worldWidth = getWidth(map.getView().getProjection().getExtent());
        const offset = Math.floor(map.getView().getCenter()[0] / worldWidth);

        // directly draw the lines with the vector context
        // 直接使用 vector context 在当前世界与相邻世界中绘制航迹
        currentLine.translate(offset * worldWidth, 0);
        vectorContext.drawGeometry(currentLine);
        currentLine.translate(worldWidth, 0);
        vectorContext.drawGeometry(currentLine);
      }
    }
  }
  // tell OpenLayers to continue the animation
  // 请求下一帧重绘继续推进动画
  map.render();
}

/**
 * Stagger feature addition with timeout.
 * 阶梯延迟注入要素到数据源中。
 * @param {Array<Feature>} features Feature array.
 * @param {number} timeout Timeout in ms.
 */
function addLater(features, timeout) {
  window.setTimeout(function () {
    let start = Date.now();
    features.forEach(function (feature) {
      feature.set('start', start);
      flightsSource.addFeature(feature);
      const duration =
        (feature.getGeometry().getCoordinates().length - 1) / pointsPerMs;
      start += duration;
    });
  }, timeout);
}
