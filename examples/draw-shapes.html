Draw Shapes / 绘制规则图形

draw18 edit13 freehand3 vector87

Using the ol/interaction/Draw to create regular shapes / 使用 ol/interaction/Draw 交互创建规则几何图形（正多边形、矩形、多角星）。

This demonstrates the use of the geometryFunction option for the ol/interaction/Draw. Select a shape type from the dropdown above to start drawing. To activate freehand drawing, hold the Shift key. Square drawing is achieved by using type: 'Circle' type with a geometryFunction that creates a 4-sided regular polygon instead of a circle. Box drawing uses type: 'Circle' with a geometryFunction that creates a box-shaped polygon instead of a circle. Star drawing uses a custom geometry function that converts a circle into a star using the center and radius provided by the draw interaction.

本示例演示如何利用 ol/interaction/Draw 的 geometryFunction（几何自定义转换函数）选项来绘制各类规则多边形。从下拉菜单中选择形状即可开始绘制；按住 Shift 键可激活自由手绘。

  • 正方形（Square）：基于 type: 'Circle'，配合 createRegularPolygon(4) 生成 4 条边的正多边形；
  • 矩形边界框（Box）：基于 type: 'Circle'，配合 createBox() 将起始点与当前点转换为轴对齐的矩形面；
  • 多角星（Star）：通过自定义几何函数，根据起始圆心与当前半径动态按角度计算交替长短径顶点，生成五角星/多角星面要素。

📚 学习记录与技术深度解读 (Learning Notes)

1. 为什么“正方形/矩形/星形”的基础类型都是 type: 'Circle'？

  • 双参数手势抽象：圆（Circle）在交互本质上是由“圆心点（Center）+ 拖拽边缘半径点（Radius Point）”这两个离散坐标所唯一定义的。
  • 规则几何的天然映射：无论是正方形、正六边形还是拉框矩形，它们都完美匹配“中心-角点”或“对角两点”的双点交互模型。OpenLayers 巧妙复用了 Circle 交互的底层手势跟踪机制，省去了重新编写拖拽状态机的重复开销。

2. geometryFunction 的函数签名与性能机制

  • 签名：function(coordinates, geometry) => geometry
    • coordinates：当前正在绘制的临时点位数组（例如 [center, currentPoint]）；
    • geometry：上一次调用返回的可重用几何对象实例。
  • 避免 GC 抖动（内存优化）：在鼠标移动的 60fps 渲染循环中，优先原地调用 geometry.setCoordinates(...) 更新坐标，而不是反复 new Polygon()，可大幅降低垃圾回收器（Garbage Collection）压力，保证绘制手势极为丝滑。
main.js 
import Map from 'ol/Map.js';
import View from 'ol/View.js';
import Polygon from 'ol/geom/Polygon.js';
import Draw, {createBox, createRegularPolygon} from 'ol/interaction/Draw.js';
import TileLayer from 'ol/layer/Tile.js';
import VectorLayer from 'ol/layer/Vector.js';
import OSM from 'ol/source/OSM.js';
import VectorSource from 'ol/source/Vector.js';

const raster = new TileLayer({
  source: new OSM(),
});

const source = new VectorSource({wrapX: false});

const vector = new VectorLayer({
  source: source,
});

const map = new Map({
  layers: [raster, vector],
  target: 'map',
  view: new View({
    center: [-11000000, 4600000],
    zoom: 4,
  }),
});

const typeSelect = document.getElementById('type');

let draw; // global so we can remove it later / 全局变量，便于后续动态移除与重置

function addInteraction() {
  let value = typeSelect.value;
  if (value !== 'None') {
    let geometryFunction;
    if (value === 'Square') {
      value = 'Circle';
      // Generate 4-sided regular polygon (square)
      // 生成 4 边正多边形（正方形）
      geometryFunction = createRegularPolygon(4);
    } else if (value === 'Box') {
      value = 'Circle';
      // Generate axis-aligned bounding box from two points
      // 将两点对角线转换为矩形面
      geometryFunction = createBox();
    } else if (value === 'Star') {
      value = 'Circle';
      // Custom geometry function to construct a star polygon
      // 自定义几何函数：根据圆心与半径推算多角星交替顶点
      geometryFunction = function (coordinates, geometry) {
        const center = coordinates[0];
        const last = coordinates[coordinates.length - 1];
        const dx = center[0] - last[0];
        const dy = center[1] - last[1];
        const radius = Math.sqrt(dx * dx + dy * dy);
        const rotation = Math.atan2(dy, dx);
        const newCoordinates = [];
        const numPoints = 12;
        for (let i = 0; i < numPoints; ++i) {
          const angle = rotation + (i * 2 * Math.PI) / numPoints;
          const fraction = i % 2 === 0 ? 1 : 0.5;
          const offsetX = radius * fraction * Math.cos(angle);
          const offsetY = radius * fraction * Math.sin(angle);
          newCoordinates.push([center[0] + offsetX, center[1] + offsetY]);
        }
        newCoordinates.push(newCoordinates[0].slice());
        if (!geometry) {
          geometry = new Polygon([newCoordinates]);
        } else {
          geometry.setCoordinates([newCoordinates]);
        }
        return geometry;
      };
    }
    draw = new Draw({
      source: source,
      type: value,
      geometryFunction: geometryFunction,
    });
    map.addInteraction(draw);
  }
}

/**
 * Handle shape type change event.
 * 处理形状类型下拉框切换事件。
 */
typeSelect.onchange = function () {
  map.removeInteraction(draw);
  addInteraction();
};

document.getElementById('undo').addEventListener('click', function () {
  draw.removeLastPoint();
});

addInteraction();
