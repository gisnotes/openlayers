Image Filters / 图像滤镜

filter3 imagemanipulation3 maptiler26

Apply a filter to imagery / 对遥感影像图层应用卷积图像滤镜（锐化、模糊、浮雕、边缘检测等）。

Layer rendering can be manipulated in prerender and postrender event listeners. These listeners get an event with a reference to the Canvas rendering context. In this example, the postrender listener applies a filter to the image data.

图层的渲染管线可通过 prerender 和 postrender 事件监听器进行拦截和后处理。这些事件对象中挂载了当前图层的 Canvas 2D 绘图上下文（event.context）。本示例中，通过在 postrender 监听器中获取视口像素数据并应用自定义 3×3 卷积核（Convolution Kernel），实现了锐化、模糊、浮雕、投影和边缘检测等图像后处理特效。

📚 学习记录与技术深度解读 (Learning Notes)

1. 渲染事件拦截与 Canvas 上下文 (Render Hooks & Canvas Context)

  • 后处理拦截机制: 当切片图层触发 postrender 事件时，当前视口范围内的切片已全部栅格化绘制到图层独享或共享的 HTML5 Canvas 画布上。此时挂载于事件参数中的 event.context 允许开发者直接调用原生 2D Canvas 上下文方法。
  • 像素级数据读写: 通过 context.getImageData(0, 0, width, height) 可以提取视口当前帧的 RGBA 原始字节流（Uint8ClampedArray），并在经过数学变换后通过 context.putImageData(output, 0, 0) 将新像素阵列写回，实现任意基于 CPU 的着色器后处理效果。

2. 图像空间卷积核与亮度归一化原理 (Convolution & Normalization)

  • 二维卷积数学模型: 卷积滤波将目标像素周围 $3 \times 3$ 邻域内的各点颜色通道与预设矩阵（Kernel）相乘并累加： $$I_{\text{out}}(x, y) = \sum_{j=-1}^{1} \sum_{i=-1}^{1} I_{\text{in}}(x + i, y + j) \cdot K(i + 1, j + 1)$$
  • 权重归一化 (Kernel Normalization): 当卷积核中各权值之和大于 0 时（如模糊滤波 $\sum K = 9$），必须将所有权重除以该总和（每个权重为 $1/9$），以确保滤波后的画面整体能量守恒（即总亮度不发生漂移）；对于差分求导类算子（如边缘检测核 $\sum K = 0$），则无需除以总和，且 Alpha 通道强制设置为 255 保证完全不透明。

3. 生产实践陷阱与避坑指南 (Production Pitfalls & Best Practices)

  • CPU 计算瓶颈与 GPU 着色器选型: getImageData() 与纯 JS 的双重像素循环在常规 1080P/2K/4K 屏幕上计算量巨大（通常需数十毫秒），会导致地图拖拽漫游时出现严重掉帧。生产环境中若需全局图像后处理，首选 OpenLayers 的 WebGLTile 搭配自定义片元着色器（Fragment Shader），将矩阵卷积计算交给 GPU 硬件多核心并行管线处理。
  • 跨域画布污染 (CORS Taint): 若切片源服务器未在 HTTP 响应头中正确下发 Access-Control-Allow-Origin: *，浏览器会标记 Canvas 为被污染状态，调用 getImageData() 将直接抛出 SecurityError 安全异常。在配置图层源时务必确保切片支持 CORS。
  • 边界箝位与越界保护 (Clamp to Edge): 视口外边缘像素缺少相邻采样点，代码采用 Math.min(maxCoord, Math.max(0, currentCoord)) 实施边缘箝位采样，防止数组越界计算产生黑边或内存访问错误。
main.js 
import Map from 'ol/Map.js';
import View from 'ol/View.js';
import TileLayer from 'ol/layer/Tile.js';
import {fromLonLat} from 'ol/proj.js';
import ImageTile from 'ol/source/ImageTile.js';

// MapTiler API Key for satellite imagery access
// MapTiler API 密钥：用于加载高分辨率卫星影像瓦片
const key = 'Get your own API key at https://www.maptiler.com/cloud/';
const attributions =
  '<a href="https://www.maptiler.com/copyright/" target="_blank">&copy; MapTiler</a> ' +
  '<a href="https://www.openstreetmap.org/copyright" target="_blank">&copy; OpenStreetMap contributors</a>';

// Satellite imagery tile layer
// 卫星遥感切片图层
const imagery = new TileLayer({
  source: new ImageTile({
    attributions: attributions,
    url: 'https://api.maptiler.com/maps/satellite/{z}/{x}/{y}.jpg?key=' + key,
    tileSize: 512,
    maxZoom: 20,
  }),
});

// Initialize map centered on North America Pacific Northwest
// 初始化地图实例：中心定位至北美太平洋西北地区
const map = new Map({
  layers: [imagery],
  target: 'map',
  view: new View({
    center: fromLonLat([-120, 50]),
    zoom: 6,
  }),
});

// 3x3 Convolution kernels for various image processing effects
// 预定义的 3x3 空间卷积核字典：涵盖直通、锐化、高斯模糊、阴影投影、浮雕及边缘检测
const kernels = {
  none: [0, 0, 0, 0, 1, 0, 0, 0, 0],
  sharpen: [0, -1, 0, -1, 5, -1, 0, -1, 0],
  sharpenless: [0, -1, 0, -1, 10, -1, 0, -1, 0],
  blur: [1, 1, 1, 1, 1, 1, 1, 1, 1],
  shadow: [1, 2, 1, 0, 1, 0, -1, -2, -1],
  emboss: [-2, 1, 0, -1, 1, 1, 0, 1, 2],
  edge: [0, 1, 0, 1, -4, 1, 0, 1, 0],
};

// Normalize kernel weights so that overall image brightness is preserved
// 卷积核归一化函数：当权重和大于 0 时平分权重保持整体亮度，否则标记未归一化（如边缘检测）
function normalize(kernel) {
  const len = kernel.length;
  const normal = new Array(len);
  let i,
    sum = 0;
  for (i = 0; i < len; ++i) {
    sum += kernel[i];
  }
  if (sum <= 0) {
    normal.normalized = false;
    sum = 1;
  } else {
    normal.normalized = true;
  }
  for (i = 0; i < len; ++i) {
    normal[i] = kernel[i] / sum;
  }
  return normal;
}

const select = document.getElementById('kernel');
let selectedKernel = normalize(kernels[select.value]);

/**
 * Update the kernel and re-render on change.
 * 监听滤镜下拉选单变更：重新计算归一化卷积核并触发地图画布重绘
 */
select.onchange = function () {
  selectedKernel = normalize(kernels[select.value]);
  map.render();
};

/**
 * Apply a filter on "postrender" events.
 * 监听遥感图层的 postrender 事件：拦截栅格化后的 Canvas 像素应用空间卷积
 */
imagery.on('postrender', function (event) {
  convolve(event.context, selectedKernel);
});

/**
 * Apply a convolution kernel to canvas.  This works for any size kernel, but
 * performance starts degrading above 3 x 3.
 * 对 Canvas 2D 绘图上下文应用空间卷积核：遍历每个像素的 3x3 邻域并加权合成
 * @param {CanvasRenderingContext2D} context Canvas 2d context.
 * @param {Array<number>} kernel Kernel.
 */
function convolve(context, kernel) {
  const canvas = context.canvas;
  const width = canvas.width;
  const height = canvas.height;

  const size = Math.sqrt(kernel.length);
  const half = Math.floor(size / 2);

  // Read raw RGBA pixel byte stream from canvas context
  // 从 Canvas 2D 上下文中读取原始 RGBA 像素字节数组
  const inputData = context.getImageData(0, 0, width, height).data;

  // Allocate empty target image buffer
  // 分配同样尺寸的目标像素缓冲区
  const output = context.createImageData(width, height);
  const outputData = output.data;

  // Iterate over every pixel on the canvas
  // 逐行、逐列遍历画布上的每一个像素点
  for (let pixelY = 0; pixelY < height; ++pixelY) {
    const pixelsAbove = pixelY * width;
    for (let pixelX = 0; pixelX < width; ++pixelX) {
      let r = 0,
        g = 0,
        b = 0,
        a = 0;
      // Convolve surrounding neighborhood window
      // 遍历当前像素的 3x3 邻域窗口并乘以对应核权重
      for (let kernelY = 0; kernelY < size; ++kernelY) {
        for (let kernelX = 0; kernelX < size; ++kernelX) {
          const weight = kernel[kernelY * size + kernelX];
          // Clamp coordinates to image boundaries
          // 边界箝位保护：防止采样坐标越界
          const neighborY = Math.min(
            height - 1,
            Math.max(0, pixelY + kernelY - half),
          );
          const neighborX = Math.min(
            width - 1,
            Math.max(0, pixelX + kernelX - half),
          );
          const inputIndex = (neighborY * width + neighborX) * 4;
          r += inputData[inputIndex] * weight;
          g += inputData[inputIndex + 1] * weight;
          b += inputData[inputIndex + 2] * weight;
          a += inputData[inputIndex + 3] * weight;
        }
      }
      const outputIndex = (pixelsAbove + pixelX) * 4;
      outputData[outputIndex] = r;
      outputData[outputIndex + 1] = g;
      outputData[outputIndex + 2] = b;
      outputData[outputIndex + 3] = kernel.normalized ? a : 255;
    }
  }
  // Write modified pixels back to the Canvas context
  // 将后处理完成的像素写回当前画布
  context.putImageData(output, 0, 0);
}
